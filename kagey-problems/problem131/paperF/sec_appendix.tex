\section{Classical inputs}\label{app:classical}

This appendix gives the proofs of Lemmas~\ref{lem:rv} and~\ref{lem:profile}, four
classical facts on sums of runs, and the bounds on the kernels of one sign that
enter the proof of Theorem~\ref{thm:bulk}.

\subsection{Regular variation and the profile}\label{app:rv}

\begin{proof}[Proof of Lemma~\ref{lem:rv}]
Parts (i) to (iii) are the cited theorems, and $T_\sgn(x/c)\le C_cT_\sgn(x)$ follows
from (i) for large $x$ and from positivity for the others. For (iv), $nT_\sgn(n)$ has
index $1-\alpha_\sgn>-1$ and $T_\sgn$ has index $-\alpha_\sgn<-1$, so the cited
propositions give the asymptotics, and positivity gives the bounds for all $w\ge1$.
For (v), $B_N$ is nondecreasing since each $a_\sgn$ is increasing. If
$\alpha_\plus\ne\alpha_\minus$, then $B_N=b_{\sgn,N}$ for the sign with the smaller index
and all large $N$. If the indices are equal, $B_{cN}/B_N$ lies between the two ratios
$b_{\sgn,cN}/b_{\sgn,N}$, which tend to $c^{1/\alpha_*}$. So $B_NT_\sgn(N)$ has the
negative index $1/\alpha_*-\alpha_\sgn$ and tends to
zero~\cite[Proposition~1.5.1]{bgt1989}, the definition of $a_\sgn$ gives
$NT_\sgn(b_{\sgn,N})\to s$, and $B_N\log N=o(N)$ since $1/\alpha_*<1$. For (vi), take
$\delta=(\alpha_\sgn-1)/2$ in (ii) and use $s/2\le NT_\sgn(b_{\sgn,N})\le2s$ for large $N$.
\end{proof}

\begin{proof}[Proof of Lemma~\ref{lem:profile}]
Under (RV), $0<p<1$. For $1<\alpha<2$ the real parts of $(-it)^\alpha$ and $(it)^\alpha$
equal $|t|^\alpha\cos\frac{\pi\alpha}2<0$, and $(-ict)^\alpha=c^\alpha(-it)^\alpha$ for $c\ge0$.
So $W_\theta$ has the characteristic function
\[
 \widehat h_\theta(t)=\exp\Bigl(K_{\alpha_\plus}\bigl((1-p)\theta_\plus\bigr)^{\alpha_\plus}(-it)^{\alpha_\plus}
 +K_{\alpha_\minus}\bigl(p\,\theta_\minus\bigr)^{\alpha_\minus}(it)^{\alpha_\minus}\Bigr),
\]
continuous in $\theta$. One coordinate of $\theta\in\Theta$ equals $1$, so
$|\widehat h_\theta|\le\Phi$ with $\Phi(t)=1$ for $|t|<1$ and $\Phi(t)=e^{-c|t|^{\alpha_*}}$
otherwise, where $c=\min_\sgn K_{\alpha_\sgn}|\cos\frac{\pi\alpha_\sgn}2|\min(p,1-p)^{\alpha_\sgn}>0$.
Fourier inversion gives (a) with $C_1=\frac1{2\pi}\int\Phi$ and
$C_2=\frac1{2\pi}\int|t|\Phi(t)\,dt$, and dominated convergence gives the continuity
in (c). If $N'/N\to1$, then $\theta_{N'}-\theta_N\to0$ by Lemma~\ref{lem:rv}(i), and
uniform continuity on the compact $\Theta$ gives the rest of (c).

For (b), the set $\{h_\theta:\theta\in\Theta\}$ is compact in the uniform norm and
each $h_\theta$ vanishes at infinity, so a finite net gives $\varepsilon_h(R)\to0$. If
$\varepsilon'_h(R)\not\to0$, there are $\varepsilon>0$, $R_i\to\infty$ and $N_i$ with
$\int_{|x|>R_i}h_{N_i}\ge\varepsilon$ and, by compactness, $\theta_{N_i}\to\theta$. Then
$h_{N_i}\to h_\theta$ in $L^1$ by Scheff\'e's lemma, which is a contradiction.

For (d), fix a sign and write $Z=Z_\sgn$, $g=g_\sgn$, $\alpha=\alpha_\sgn$. A sum of $n$
independent copies of $Z$ has the law of $n^{1/\alpha}Z$. If $Z\ge0$ almost surely,
then the strong law of large numbers applied to $\min(Z_i,1)$ makes
$n^{-1/\alpha}(Z_1+\dots+Z_n)\to\infty$, which contradicts tightness, and the same holds
for $-Z$. So the open set $U=\{g>0\}$ meets both half lines. The density $g^{*n}$ is
continuous, positive at $y_1+\dots+y_n$ when every $y_i\in U$, and equal to
$n^{-1/\alpha}g(n^{-1/\alpha}\,\cdot)$. Hence $(a-\varepsilon,a+\varepsilon)\subset U$ with
$0<\varepsilon<a$ gives $c_n(a-\varepsilon,a+\varepsilon)\subset U$ with
$c_n=n^{1-1/\alpha}$, and since $c_{n+1}/c_n\to1$ these intervals cover a half line. The
same holds on the other side, and the case $n=2$ gives $U=\R$. If $\theta_\sgn=1$, then
$W_\theta=aZ_\sgn+Y$ with $a\ne0$ and $Y$ independent, so $h_\theta(x)=\E[|a|^{-1}g_\sgn((x-Y)/a)]>0$.
The two minima in (d) are minima of positive continuous functions on $\Theta$.

For (e) with $\tau=\plus$, condition on $Z_\minus=y$ and use $1-p=\mu_\minus/s$ and
$p/(1-p)=\mu_\plus/\mu_\minus$. For $\tau=\minus$ condition on $Z_\plus$.
\end{proof}

\subsection{Sums of runs}\label{sec:bulk-inputs}

We write $\bar T_\sgn(x)=\Prob(X_\sgn>x)\sim T_\sgn(x)$.

\begin{lemma}[stable and local limits]\label{lem:stable}
Assume (RV) and (SPAN) for the sign $\sgn$, and write $\alpha=\alpha_\sgn$,
$\mu=\mu_\sgn$, $a=a_\sgn$, $g=g_\sgn$, $T=T_\sgn$ and $S_m=S^\sgn_m$. Parts
(i), (iii) and (iv) need only (RV).
\begin{enumerate}[(i)]
\item $Z_m=(S_m-m\mu)/a(m)$ converges in law to $Z_\sgn$, whose density $g$ is
bounded, uniformly continuous, and tends to zero at infinity. Moreover
$m\bigl(\E e^{-(X_\sgn-\mu)/a(m)}-1\bigr)\to K_\alpha$.
\item (Gnedenko's local limit theorem)
$\sup_{k\in\Z}\bigl|a(m)\Prob(S_m=k)-g\bigl((k-m\mu)/a(m)\bigr)\bigr|\to0$ as $m\to\infty$,
and $\bar u_\sgn=\sup_{m\ge1}\sup_ka(m)\Prob(S_m=k)<\infty$.
\item (local large deviations) There is a constant $C_{\mathrm{LD}}=C_{\mathrm{LD}}(\sgn)$ such that
$\Prob(S_m=j)\le C_{\mathrm{LD}}\,m\,T(j-m\mu)/a(m)$
for all $m\ge1$ and all integers $j$ with $j-m\mu\ge1$.
\item (left deviations) The constant $C_-=C_-(\sgn)=\sup_{m\ge1}m\log\E e^{-(X_\sgn-\mu)/a(m)}$
is finite, and
$\Prob(S_m\le m\mu-y)\le e^{C_-}\,e^{-y/a(m)}$ for all $m\ge1$ and $y\ge0$.
\end{enumerate}
\end{lemma}

\begin{proof}
(i) The domain of attraction is classical~\cite[Section~XVII.5]{feller1971}, and we
check the constant $K_\alpha$. Let $X=X_\sgn$ and
$\Lambda(\xi)=\E[e^{-\xi X}-1+\xi X]$ for $\operatorname{Re}\xi\ge0$. Fubini's theorem gives,
for $\xi=te^{i\vartheta}$ with $t>0$ and $|\vartheta|\le\pi/2$,
\[
 \Lambda(\xi)=\int_0^\infty \xi\bigl(1-e^{-\xi y}\bigr)\bar T(y)\,dy
 =\int_0^\infty e^{i\vartheta}\bigl(1-e^{-e^{i\vartheta}v}\bigr)\bar T(v/t)\,dv .
\]
The integrand is at most $\min(2,v)\bar T(v/t)$ in modulus, and
$\bar T(v/t)/\bar T(1/t)\to v^{-\alpha}$ as $t\to0$, dominated by
Lemma~\ref{lem:rv}(ii) for $v\ge tx_\delta$ with $1<\alpha\pm\delta<2$, while
$v<tx_\delta$ contributes $o(\bar T(1/t))$ since $\alpha<2$. So, uniformly in
$\vartheta$, $\Lambda(te^{i\vartheta})=\bar T(1/t)(F(e^{i\vartheta})+o(1))$ with
$F(\xi)=\int_0^\infty\xi(1-e^{-\xi v})v^{-\alpha}dv$. The function $F$ is continuous on
$\operatorname{Re}\xi\ge0$, analytic inside and homogeneous of degree $\alpha$, and an
integration by parts gives $F(1)=\Gamma(2-\alpha)/(\alpha-1)=K_\alpha$, so
$F(\xi)=K_\alpha\xi^\alpha$. With $\E e^{-\xi(X-\mu)}-1=\Lambda(\xi)+O(|\xi|^2)$,
$\xi=-i\theta/a(m)$, $m\bar T(a(m)/|\theta|)\to|\theta|^\alpha$ and $m/a(m)^2\to0$ we get
$m(\E e^{i\theta(X-\mu)/a(m)}-1)\to K_\alpha(-i\theta)^\alpha$, and L\'evy's continuity
theorem gives $Z_m\to Z_\sgn$. The limit characteristic function is integrable, which
gives the properties of $g$. With $\xi=1/a(m)$ the same expansion gives the last claim.

(ii) By (SPAN) the maximal span of $X_\sgn$ is $1$, and Gnedenko's
theorem~\cite[Theorem~4.2.1]{ibragimovlinnik1971} (see also~\cite[Theorem~8.4.1]{bgt1989})
gives the limit, with any norming $a_m\sim a(m)$. The bound $\bar u_\sgn$ follows.

(iii) This is~\cite[Theorem~1.1, bound~(1.6)]{caravennadoney2019} for the centered
variable $X_\sgn-\mu$, with $A=1/\bar T$, the interval $(-1,0]$ and $x=j-m\mu$. Its
right tail is $\bar T(x)(1+o(1))$ and its left tail vanishes for $x>\mu$. The bound
needs no lattice assumption, its norming sequence is asymptotic to
$a(m)$~\cite[Theorem~1.5.12]{bgt1989}, and $\bar T\le T$.

(iv) By (i), $m(\E e^{-(X_\sgn-\mu)/a(m)}-1)$ is bounded, and $\log(1+z)\le z$ makes $C_-$
finite. Markov's inequality for $e^{-(S_m-m\mu)/a(m)}$ gives the bound.
\end{proof}

\subsection{Kernels of one sign}\label{sec:bulk-kernels}

For each sign $\sgn$ the fresh identities use the kernels $u^\sgn_m$, $q^\sgn_m$ and
$q^\sgn_{m+1}$ ($m\ge1$), with weights $c_{\mathsf k}=1$, $\mu_\sgn$ and $\mu_\sgn$. A
kernel of sign $\sgn$ is one of these sequences, written $\mathsf k_m$. We use four
facts.
\begin{enumerate}[(K1)]
\item $\sum_{m\ge1}u^\sgn_m(j)\le1$, $\sum_{m\ge1}q^\sgn_m(j)=1$ and
$\sum_{m\ge1}q^\sgn_{m+1}(j)\le1$ for $j\ge1$, by~\eqref{eq:kernel-sums}.
\item $\mathsf k_m(j)=0$ for $m>j$, because $S^\sgn_m\ge m$.
\item Let $J_\sgn(j)=\min\{m\ge1:S^\sgn_m\ge j\}$. Then $\Prob(J_\sgn(j)=m)=q^\sgn_m(j)$
by~\eqref{eq:q-cover}, and $u^\sgn_m(j)\le q^\sgn_m(j)$, since $\{S^\sgn_m=j\}\subset\{J_\sgn(j)=m\}$.
\item $\sum_jq^\sgn_m(j)=\mu_\sgn$ and $\sum_ju^\sgn_m(j)=1$.
\end{enumerate}

\begin{lemma}[kernel local limits]\label{lem:kernels}
Assume (RV) and (SPAN). For each sign $\sgn$ and each kernel $\mathsf k$ of sign $\sgn$,
\[
 \delta_{\mathsf k}(m)=\sup_{j\in\Z}\bigl|a_\sgn(m)\mathsf k_m(j)-c_{\mathsf k}\,g_\sgn\bigl((j-m\mu_\sgn)/a_\sgn(m)\bigr)\bigr|\to0
 \qquad(m\to\infty),
\]
and $\mathsf k_m(j)\le C_3/a_\sgn(m)$ for all $m\ge1$ and $j$, where
$C_3=C_3(\sgn)=\max\bigl(a_\sgn(1),\,\mu_\sgn r_\sgn\bar u_\sgn\bigr)$ and
$r_\sgn=\sup_{m\ge2}a_\sgn(m)/a_\sgn(m-1)<\infty$.
\end{lemma}

\begin{proof}
For $u^\sgn_m$ this is Lemma~\ref{lem:stable}(ii). The kernel $q^\sgn_m$ is a sum of
shifts of $u^\sgn_{m-1}$ by $v\ge1$ with weights $T_\sgn(v)$ of total mass $\mu_\sgn$, and
$q^\sgn_{m+1}$ is the same sum built on $u^\sgn_m$. This gives the bounds, with
$q^\sgn_1=T_\sgn\le1$. For the limit of $q^\sgn_m$ write $a=a_\sgn$, $g=g_\sgn$ and
$z=(j-m\mu)/a(m)$, and fix $v_0$ and $Z$. The shifts $v>v_0$ contribute at most
$(\bar u_\sgn+\|g\|_\infty)\sum_{v>v_0}T(v)$ to $|a(m-1)q_m(j)-\mu g(z)|$. For $v\le v_0$
the scaled point moves by at most $Z|a(m)/a(m-1)-1|+(v_0+\mu)/a(m-1)$ when $|z|\le Z$,
and stays beyond $Z/2$ in modulus for large $m$ when $|z|>Z$. So the uniform continuity
of $g$, its decay and Lemma~\ref{lem:stable}(ii) give the limit as $m\to\infty$, then
$v_0,Z\to\infty$. The kernel $q^\sgn_{m+1}$ is the same.
\end{proof}

\begin{lemma}[small counts]\label{lem:small}
Assume (RV). For each sign $\sgn$ there is $C_4=C_4(\sgn)$ such that every kernel
of sign $\sgn$ satisfies
\[
 \mathsf k_m(j)\le C_4\bigl(1+m/a_\sgn(m)\bigr)T_\sgn(j)\qquad\text{whenever }8m\mu_\sgn\le j .
\]
Consequently, for $0<\beta\le1$ there is $C_5=C_5(\sgn,\beta)$ with
\begin{equation}\label{eq:bulk-small}
 \mathsf k_m(j)\le C_5\,e_{\sgn,N},\qquad e_{\sgn,N}=\bigl(1+N/a_\sgn(N)\bigr)T_\sgn(N),
\end{equation}
for all $N\ge1$, all $\beta N\le j\le N$ and all $m$ with $8m\mu_\sgn\le\beta N$.
Moreover $B_Ne_{\sgn,N}\to0$.
\end{lemma}

\begin{proof}
Write $T=T_\sgn$, $a=a_\sgn$, $\mu=\mu_\sgn$, and let $T(x/2)\le C_TT(x)$. If
$4m\mu\le i$, then $i-m\mu\ge i/2\ge1$, and Lemma~\ref{lem:stable}(iii) gives
$u_m(i)\le C_{\mathrm{LD}}C_T\,m\,T(i)/a(m)$, which is the claim for $u_m$. Let
$8m\mu\le j$ and $m\ge2$, and split $q_m(j)=\sum_vT(v)u_{m-1}(j-v)$ at $v=j/2$. For
$v\le j/2$ the bound for $u_{m-1}$ applies at $j-v\ge4(m-1)\mu$, and the weights add up
to at most $\mu$. For $v>j/2$ we have $T(v)\le C_TT(j)$, and the values $u_{m-1}(j-v)$
add up to at most $1$. This gives the claim with $C_4=C_T+\mu C_{\mathrm{LD}}C_T^2r_\sgn$,
and $q_{m+1}$ is the same. For~\eqref{eq:bulk-small}, $m/a(m)\le C_PN/a(N)$ for $m\le N$
by Lemma~\ref{lem:rv}(iii), and $T(j)\le C_{T,\beta}T(N)$. Finally
$B_NNT_\sgn(N)/a_\sgn(N)$ has the index
$1/\alpha_*-1/\alpha_\sgn+1-\alpha_\sgn\le1-\alpha_*<0$, since
$1/\alpha_*-1/\alpha_\sgn\le\alpha_\sgn-\alpha_*$, and $B_NT_\sgn(N)\to0$.
\end{proof}
