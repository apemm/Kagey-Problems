\section{The large-concentration limit of the direct jump}\label{app:hdp}

We prove Proposition~\ref{prop:directjump}(b), with $B_1,B_2,B_3$ independent
$\mathrm{Beta}(a,a)$ as in part (a) and $a=\omega/3\ge1$. As $D$ is a thin
tube around the cyclic kernels, we expect
$\Prob_\omega(D)\approx a^{3/2}\int t^4e^{-cat^2}dt\approx1/a$.

Put $\ell_j=\log(B_j/(1-B_j))$, $g(y)=2\log\cosh(y/2)$ and $g_j=g(\ell_j)$.
Then $g$ is even and convex, $g\ge0=g(0)=g'(0)$,
$B_j(1-B_j)=\frac14e^{-g_j}$, $\log(2B_j)=\frac12(\ell_j-g_j)$ and
$\log(2(1-B_j))=-\frac12(\ell_j+g_j)$. As $dB_j/d\ell_j=B_j(1-B_j)$, $\ell_j$
has density $e^{-ag(\ell_j)}/Z$ with $Z=2\sqrt\pi\Gamma(a)/\Gamma(a+\frac12)$.
So $D$ is
\begin{equation}\label{eq:hdp-constraints}
 M_1\le g_1+g_2,\qquad M_2\le g_1+g_3,\qquad M_3\le g_2+g_3,
\end{equation}
with $M_1=\ell_1+\ell_2$, $M_2=\ell_3-\ell_1$, $M_3=-\ell_2-\ell_3=-M_1-M_2$.
With $\psi=g_1+g_2+g_3$,
\begin{equation}\label{eq:hdp-integral}
 \Prob_\omega(D)=Z^{-3}\int_De^{-a\psi(\ell)}d\ell.
\end{equation}
For real $y$,
\begin{align}
 \frac{y^2}4-\frac{y^4}{96}\le g(y)&\le\frac{y^2}4,\tag{g1}\label{eq:hdp-g1}\\
 g(y)&\ge\frac1{10}\min(y^2,|y|).\tag{g2}\label{eq:hdp-g2}
\end{align}
With $u=y/2$, the upper bound is $\cosh u\le e^{u^2/2}$
(termwise), and the lower bound holds as
$\chi=\log\cosh u-\frac{u^2}2+\frac{u^4}{12}$ has $\chi(0)=\chi'(0)=0$ and
$\chi''=u^2-\tanh^2u\ge0$. For \eqref{eq:hdp-g2},
$g''=\frac12\mathrm{sech}^2(y/2)>0.2$ on $|y|\le2$, and for $|y|\ge2$
convexity gives $g(y)\ge\frac12|y|g(2)>0.43|y|$. Let $v=(1,-1,1)$ and $t=\langle\ell,v\rangle/3$. The map
$\ell\mapsto(t,M_1,M_2)$ has determinant $1$ and inverse $\ell=tv+w$ with
$w=\frac13(M_1-M_2,2M_1+M_2,M_1+2M_2)\perp v$. So $|\ell|^2=3t^2+|w|^2$ with
$|w|^2=\frac13(M_1^2+M_2^2+M_3^2)$, and with $|M|$, $|M|_\infty$ the 2-norm
and max norm of $(M_1,M_2)$,
\begin{equation}\label{eq:hdp-w}
 \tfrac13|M|^2\le|w|^2\le|M|^2\le2|M|_\infty^2.
\end{equation}

\begin{proof}[Proof of Proposition~\ref{prop:directjump}(b)]
First, on $D$, $M_1\le\psi$ and $M_1=-M_2-M_3\ge-2\psi$, and the same
for $M_2$. So by \eqref{eq:hdp-g1}
\begin{equation}\label{eq:hdp-apriori}
 |M|_\infty\le2\psi(\ell)\le\tfrac12|\ell|^2=\tfrac12(3t^2+|w|^2).
\end{equation}
Next let $\varepsilon=1/12$ and
$U=\{|t|\le\varepsilon,\ |M|_\infty\le\varepsilon\}$. Off $U$,
$|\ell|\ge\max(\sqrt3|t|,|M|/\sqrt3)\ge\varepsilon/\sqrt3$, and some
$|\ell_j|\ge|\ell|/\sqrt3$. So \eqref{eq:hdp-g2} gives
$\psi\ge\gamma:=0.1\min(|\ell|^2/3,|\ell|/\sqrt3)\ge\gamma_0=\varepsilon^2/90$.
Hence $e^{-a\psi}\le e^{-(a-1)\gamma_0}e^{-\gamma}$ off $U$, and
$\int_{D\setminus U}e^{-a\psi}=O(e^{-\gamma_0a})$. On $D\cap U$, \eqref{eq:hdp-apriori} and \eqref{eq:hdp-w} give
$|M|_\infty\le\frac32t^2+\varepsilon|M|_\infty$, so $|M|_\infty\le2t^2$. On
$E=\{|t|\le\varepsilon,\ |M|_\infty\le2t^2\}$ we have $|w|^2\le8t^4$, so
$\ell_j=\pm t+w_j$ gives $|\ell_j|\le2|t|$ and $|\ell_j^2-t^2|\le7|t|^3$.
Then \eqref{eq:hdp-g1} gives $|g_j-\frac14t^2|\le2|t|^3$, and with
$|\ell|^2=3t^2+|w|^2$ it bounds $\psi$. Hence
\begin{equation}\label{eq:hdp-pair}
 \bigl|g_i+g_j-\tfrac12t^2\bigr|\le4|t|^3\quad(i\ne j),
 \qquad\bigl|\psi-\tfrac34t^2\bigr|\le2t^4.
\end{equation}
Let $\mathcal T(h)=\{M:M_1,M_2\le h,\ M_1+M_2\ge-h\}$, a triangle of area
$\frac92h^2$ in $\{|M|_\infty\le2h\}$. Let $D_t$ be the section of $D\cap U$
at $t$ and $h_\pm=\frac12t^2(1\pm8|t|)\ge0$. If $M\in D_t$, then
$(t,M)\in E$, and \eqref{eq:hdp-constraints} and \eqref{eq:hdp-pair} give
$M_1,M_2,M_3\le h_+$. If $M\in\mathcal T(h_-)$, then
$|M|_\infty\le2h_-\le\min(t^2,\varepsilon)$, so $(t,M)\in E\cap U$, and
\eqref{eq:hdp-pair} gives \eqref{eq:hdp-constraints}. So
$\mathcal T(h_-)\subseteq D_t\subseteq\mathcal T(h_+)$, and
\begin{equation}\label{eq:hdp-area}
 \tfrac98t^4(1-8|t|)^2\le|D_t|\le\tfrac98t^4(1+8|t|)^2.
\end{equation}
Finally, by Fubini (Jacobian $1$), \eqref{eq:hdp-pair} and
\eqref{eq:hdp-area}, $J=\int_{D\cap U}e^{-a\psi}$ lies between the integrals
over $|t|\le\varepsilon$ of
$\frac98t^4(1\mp8|t|)^2e^{-a(\frac34t^2\pm2t^4)}$. With $t=s/\sqrt a$ these
are $a^{-5/2}$ times integrals whose integrands tend to
$\frac98s^4e^{-3s^2/4}$ and are at most $\frac92s^4e^{-s^2/2}$, since
$8|t|<1$ and $2s^4/a\le2\varepsilon^2s^2$. By dominated convergence and
$\int s^4e^{-3s^2/4}ds=\frac34\sqrt\pi(\frac43)^{5/2}$,
$J=a^{-5/2}(\sqrt{3\pi}+o(1))$, and the same holds over $D$. As $\Gamma$ is
log-convex, $\Gamma(a+\frac12)^2\le a\Gamma(a)^2$ and
$\Gamma(a+1)^2\le(a+\frac12)\Gamma(a+\frac12)^2$, so
$Z=2\sqrt{\pi/a}(1+O(1/a))$. Then \eqref{eq:hdp-integral} gives
$\Prob_\omega(D)=\frac{\sqrt3}{8\pi a}(1+o(1))=\frac{3\sqrt3}{8\pi\omega}(1+o(1))$.
\end{proof}
