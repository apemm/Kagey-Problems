\begin{enumerate}[(1),leftmargin=*]
\item An $h\times m$ grid has $h$ rows and $m$ columns, and $V$ is its set of points.
A $2\times m$ grid is a \emph{strip}, and an $n\times n$ grid is a \emph{square}.
\item The point $(i,j)$ is in column $i$ and row $j$, with $(1,1)$ at the top left, as in
Figure~\ref{fig:game}(a).
\item A \emph{rectangle} $R$ is identified with its set $C(R)$ of four corners, which are
grid points spanning a rectangle of positive area. It is \emph{axis-aligned} if its sides
are parallel to the rows and columns, and \emph{tilted} otherwise.
\item A \emph{blackout} is a set $S\subseteq V$, and $K=V\setminus S$ is the
\emph{kept set}. The \emph{presentation} of a rectangle $R$ is the set
$\pres(R)=C(R)\setminus S$ of its white corners.
\item The \emph{black rows} of column $i$ form the set $B_i$ of rows $j$ with
$(i,j)\in S$. Column $i$ is \emph{empty}, a \emph{singleton}, a \emph{double}, or a
\emph{triple} when $|B_i|$ is $0$, $1$, $2$, or $3$. It is \emph{full} when $|B_i|=h$.
\item The \emph{pair set} $F$ is the set of row pairs contained in some $B_i$. We put
$q_h=\lfloor h^2/4\rfloor$, the quarter-square \seqnum{A002620}$(h)$.
\end{enumerate}

Swapping rows and columns gives $f(h,m)=f(m,h)$ and $a_h(m)=a_m(h)$. The tables
$f(h,m)$ and $a_h(m)$ are \seqnum{A397315} and \seqnum{A399580}, read by antidiagonals
with both indices starting at $2$. Section~\ref{sec:height} also uses width $0$ and height $1$, with no
rectangles, where $f(h,0)=0$, $a_h(0)=1$, $f(1,m)=m$, and $a_1(m)=1$.

The empty presentation is allowed, so a valid blackout hides at most one rectangle.

\begin{definition}[valid blackout]\label{def:valid}
A blackout $S$ is \emph{valid} if different rectangles always have different
presentations, that is, if $\pres(R_1)\ne\pres(R_2)$ whenever $R_1\ne R_2$. The
\emph{maximum blackout} $f(h,m)$ is the largest size of a valid blackout of the
$h\times m$ grid, and $a_h(m)$ is the number of valid blackouts of size $f(h,m)$, counted
with their coordinates.
\end{definition}

\begin{corollary}[hitting sets]\label{cor:hitting}
A blackout $S$ is valid if and only if the kept set $K$ meets
$C(R_1)\triangle C(R_2)$ for every pair of distinct rectangles $R_1$, $R_2$. So
$hm-f(h,m)$ is the least size of a set meeting every such difference set.
\end{corollary}

\begin{proof}
Apply Lemma~\ref{lem:diff} to $C(R_1)$, $C(R_2)$, and $S$ for every pair.
\end{proof}

So every subset of a valid blackout is valid, and finding no valid blackout of exactly
$k$ points proves $f(h,m)<k$.

\begin{lemma}[three corners suffice]\label{lem:threecorners}
Three corners of a rectangle determine the rectangle. So two distinct rectangles with
the same presentation both show at most two corners.
\end{lemma}

\begin{proof}
Three corners of a rectangle form a right triangle. A triangle has one right angle, at
the corner $q$ adjacent to the other two, $p$ and $r$, so the fourth corner $p+r-q$ is
determined.
\end{proof}
