\section{Introduction}\label{sec:intro}

You black out some points of the $h\times m$ grid, and we call this set the
\emph{blackout}. A game master
draws a rectangle, tilted or not, with corners on grid points, and says which corners are
white. You must name all four. Kagey~\cite{kageyG4G14,kagey001} asks for the largest
\emph{valid} blackout, where no two rectangles look the same (Definition~\ref{def:valid}). Figure~\ref{fig:game} plays one round.

Let $f(h,m)$ be the largest size of a valid blackout, and
$a_h(m)$ the number of these. Our OEIS entries \seqnum{A397315} and
\seqnum{A399580}~\cite{oeis} first recorded the conjecture $f(3,m)=m+2$, which the Omega
Institute~\cite{trureturing2026} proved in Lean, and the guess $f(h,m)=h+m$ for
$\min(h,m)\ge4$ apart from $f(4,4)=7$ and $f(4,5)=8$. The guess first fails at
$5\times8$ (Theorem~\ref{thm:fiverow}). Put $q_h=\lfloor h^2/4\rfloor$. Results that rest on
a computer check are marked computer-assisted in their statements.

\begin{introthm}[narrow grids]\label{thm:A}
\leavevmode
\begin{enumerate}[(a)]
\item We have $f(h,m)\ge h+m-1$, and $f(h,m)\le m+q_h$ if $m\ge3$.
\item We have $f(2,m)=m+1$ and $f(3,m)=m+2$ for $m\ge3$, $f(4,m)=m+4$ for $m\ge6$,
and $f(5,5)=10$.
\item (Computer-assisted, with checked certificates for $f(5,6)$ and $f(5,7)$, and the last
claim checked in Lean.) We have $f(4,4)=7$, $f(4,5)=8$,
$f(5,6)=11$, $f(5,7)=12$, and $f(5,m)=m+6$ for $m\ge8$.
\item (Computer-assisted for $m\ge9$, with checked certificates for $m\le14$ and checked in Lean
for $m\ge15$.) We have $f(6,m)\le m+8$ for $8\le m\le14$, and $f(6,m)=m+9$ for $m\ge15$.
\item (Computer-assisted, with checked certificates.) We have $f(6,6)=12$,
$f(6,7)=13$, $f(7,7)=14$, and $f(6,8)=15$.
\end{enumerate}
\end{introthm}

Part (a) is Lemma~\ref{lem:cross} and Theorem~\ref{thm:rowbound}, (b) is
Theorem~\ref{thm:strip}, Section~\ref{sec:three}, and Theorems~\ref{thm:fourvalue}
and~\ref{thm:fivefive}, (c) is Corollary~\ref{cor:fourboundary} and
Theorem~\ref{thm:fiverow}, (d) is Theorem~\ref{thm:sixrow}, and (e) is
Theorem~\ref{thm:refute}. For (c) and (d) the finite collar theorem
(Theorem~\ref{thm:collar}) reduces validity at every larger width to one finite board.

\begin{introthm}[three-row optima]\label{thm:B}
For $m\ge4$ we have $a_3(m)=4(m-1)(3m+2)\,3^{m-5}+E_m$, where $E_m>0$ counts the optima
with a column black in rows $1$ and $3$, and $E_m=O(m^2(1+\sqrt3)^m)$. So
$a_3(m)\sim\frac4{81}m^23^m$, and a fraction tending to $4/27$ of the $m^23^{m-1}$
axis-valid candidates is valid.
The series $\sum_{m\ge3}a_3(m)z^m$ is rational. (Computer-assisted.) Its reduced
denominator is $(1-3z)^3(1-2z-2z^2)^3(1+2z^2)^3$, and
$E_m\sim\frac{19+11\sqrt3}{432}m^2(1+\sqrt3)^m$.
\end{introthm}

See Theorems~\ref{thm:threecount}, \ref{thm:threegf}, and~\ref{thm:threeclosed}, and
Section~\ref{app:threegf}.

\begin{introthm}[four rows]\label{thm:C}
If $m\ge4$, no valid blackout of the $4\times m$ grid hides all corners of a
rectangle. The series $\sum_{m\ge6}a_4(m)z^m$ is rational with
reduced denominator of degree $420$, and $a_4(m)\sim c\,m^4\rho_4^m$ with
$c\approx8.1611\times10^{-5}$ and $\rho_4=3.35199\ldots$, an algebraic number of degree
$20$. (Computer-assisted.)
\end{introthm}

See Theorems~\ref{thm:hidden} and~\ref{thm:fourgfexact} and Corollary~\ref{cor:fourgrowth}.

\begin{introthm}[every fixed height]\label{thm:E}
Fix $h\ge1$ and write a blackout as the word $B_1\cdots B_m$ of the black rows of its
columns. The valid words form a regular language. If $h\ge2$, then $f(h,m)-m$ is
eventually periodic in $m$, and $\sum_{m\ge0}a_h(m)z^m$ is rational.
\end{introthm}

See Theorems~\ref{thm:regular} and~\ref{thm:periodic}.

\paragraph{Prior work.} The game asks for a test
cover~\cite{bellitto2018,foucaud2015,katona1966,moretshapiro1985}
(Corollary~\ref{cor:hitting}). Forbidding black axis-aligned rectangles is the
Zarankiewicz problem~\cite{reiman1958}. Distinguishing them forbids four- and six-cycles
in the row-column graph (Lemma~\ref{lem:girth}), and de Caen and
Sz\'ekely~\cite{decaenszekely1992,decaenszekely1997} study such graphs with unbalanced parts.
Neuwirth~\cite[Theorem~1(ii) and Proposition~17]{neuwirth2001} bounds their edges, so the row
bound is known in graph form. Our proof with Mantel's theorem~\cite{mantel1907} also finds and
counts the equality case. None of these works sees tilted rectangles. Column transfer~\cite{alanko2011,euler2005,mertens2024},
\cite[Proposition~V.6]{flajolet2009} handles fixed heights, and Theorem~1(3) of Guilhot,
Little, and Parkinson~\cite{glp2025} with Theorem~\ref{thm:regular} also gives
Theorem~\ref{thm:E}.
