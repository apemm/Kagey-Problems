\section{The row-column graph and general bounds}\label{sec:graph}

Call a blackout $S$ \emph{axis-valid} if it distinguishes the axis-aligned
rectangles, and a rectangle \emph{weak} if it shows at most two corners.
The \emph{row-column graph} $G_S$ joins row vertex $r_j$ to column vertex
$c_i$ for each black point $(i,j)$.

\begin{lemma}[axis-aligned rectangles and short cycles]\label{lem:girth}
If $G_S$ has no four-cycle and no six-cycle, then $S$ is axis-valid. The converse holds for $m\ge3$.
\end{lemma}

\begin{proof}
For the converse, let $m\ge3$. A four-cycle $r_j\,c_i\,r_{j'}\,c_{i'}$ and a third column $w$ make $\{i,w\}\times\{j,j'\}$ and $\{i',w\}\times\{j,j'\}$ show the same points. A six-cycle $r_a\,c_x\,r_b\,c_y\,r_c\,c_z$ makes $\{x,y\}\times\{a,b\}$ and $\{y,z\}\times\{a,c\}$ black except possibly at $(y,a)$.

For the first statement, let $R_1\ne R_2$ be axis-aligned with the same presentation, so $C(R_1)\triangle C(R_2)\subseteq S$ (Lemma~\ref{lem:diff}). Opposite corners determine a rectangle, so by symmetry the common corners lie in one row $a$. With none, $R_1$ is a black four-cycle. Two, $(x,a)$ and $(y,a)$, leave the other rows $b\ne c$ and the black four-cycle $r_b\,c_x\,r_c\,c_y$. One, $(x,a)$, with $R_1=\{x,y\}\times\{a,b\}$ and $R_2=\{x,z\}\times\{a,c\}$, gives the black six-cycle $r_a\,c_y\,r_b\,c_x\,r_c\,c_z$.
\end{proof}

\begin{theorem}[row bound]\label{thm:rowbound}
Let $m\ge3$. Every axis-valid blackout of the $h\times m$ grid has at most $m+q_h$ points, and at most $h+q_m$ if also $h\ge3$. Hence $f(h,m)\le m+q_h$.
\end{theorem}

For each nonempty column $i$ choose a tree $T_i$ on $B_i$, and let $H$ be their union.

\begin{lemma}[row graph]\label{lem:H}
Let $m\ge3$ and let $S$ be axis-valid. For any choice of the trees, $H$ is simple and triangle-free, and $e(H)=|S|-t$, where $t$ is the number of nonempty columns.
\end{lemma}

\begin{proof}
By Lemma~\ref{lem:girth}, $G_S$ has no four-cycle, so $H$ has no repeated edge and $e(H)=\sum(|B_i|-1)=|S|-t$. A triangle $r_ar_br_c$ cannot lie in one tree. If its edges come from two columns
$i,i'$, two of $a,b,c$ lie in $B_i\cap B_{i'}$, giving a four-cycle. If they come
from three columns, they give a six-cycle $r_a\,c_i\,r_b\,c_{i'}\,r_c\,c_{i''}$.
\end{proof}

\begin{lemma}[Mantel \cite{mantel1907}]\label{lem:mantel}
Let $H$ be a simple triangle-free graph on $h$ vertices, let a vertex of maximum degree $\Delta$ have neighbor set $A$, and let $B$ be the other $h-\Delta$ vertices, spanning $e(B)$ edges. Then
\[
e(H)=\sum_{b\in B}d(b)-e(B)\le(h-\Delta)\Delta-e(B)\le q_h.
\]
\end{lemma}

\begin{proof}
Since $A$ is independent, every edge meets $B$. Summing degrees over $B$
counts its internal edges twice, which gives the equality. The bounds follow
from $d(b)\le\Delta$ and $(h-\Delta)\Delta\le q_h$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:rowbound}]
Lemmas~\ref{lem:H} and~\ref{lem:mantel} give $|S|=t+e(H)\le m+q_h$. Only Lemma~\ref{lem:girth} uses a third column, so for $h\ge3$ the transpose gives $|S|\le h+q_m$.
\end{proof}

\begin{theorem}[equality structure]\label{thm:equality}
Let $h\ge4$ and $m\ge3$. A blackout of the $h\times m$ grid with $m+q_h$ points is axis-valid if and only if
\begin{enumerate}[(a)]
\item every column is a singleton or a double,
\item the $q_h$ doubles have distinct row pairs, and
\item these pairs form a balanced complete bipartite graph on the rows.
\end{enumerate}
There are $p_h\,(m)_{q_h}\,h^{m-q_h}$ such blackouts, where $(m)_q$ is the falling factorial and the number $p_h$ of balanced bipartitions of the rows is $\binom h{\lfloor h/2\rfloor}$, halved for even $h$.
\end{theorem}

\begin{proof}
Let $S$ be axis-valid. Equality in Lemmas~\ref{lem:H} and~\ref{lem:mantel} forces $t=m$, $e(B)=0$, and $(h-\Delta)\Delta=q_h$ with $B$ joined to all of $A$, so $H$ is as in (c). If $|B_i|\ge3$, some $v$ has two neighbors $u,w$ in $T_i$, and as $h\ge4$ the part of $v$ has another vertex $z$. Replacing $vw$ by $uw$ in $T_i$ keeps $uz$ and $wz$ and makes the triangle $uwz$, against Lemma~\ref{lem:H}. This gives (a), and the edges of $H$, the row pairs of the doubles, give (b) and (c).

Conversely, (a) and (b) rule out four-cycles, and a six-cycle $r_a\,c_x\,r_b\,c_y\,r_c\,c_z$ would need the doubles $\{a,b\}$, $\{b,c\}$, and $\{a,c\}$, a triangle in (c), so Lemma~\ref{lem:girth} applies. For the count, choose the bipartition, place its edges in distinct columns,
and choose a black row for each remaining column.
\end{proof}

\begin{corollary}[small widths]\label{cor:small-m}
Let $h\ge4$ and $3\le m<q_h$. Then $f(h,m)\le m+q_h-1$. In particular $f(5,5)\le10$.
\end{corollary}

\begin{proof}
By Theorems~\ref{thm:rowbound} and~\ref{thm:equality}, a valid blackout of $m+q_h$ points has $q_h\le m$ doubles. Here $q_5=6$.
\end{proof}

Put $W_h=1+\lfloor(h-1)^2/4\rfloor$, so $W_3=2$, $W_4=3$, and $W_5=5$.

\begin{lemma}[leftmost corner]\label{lem:leftmost}
A tilted rectangle $R$ has a unique corner $\ell=(x,r)$ in its smallest column, and a unique corner in its smallest row. The sides from $\ell$ are $(a,-b)$ and $(c,d)$ with positive integers $a,b,c,d$ and $ac=bd$, and the corners are
\begin{equation}\label{eq:tilt-param}
(x,r),\qquad(x+a,\,r-b),\qquad(x+c,\,r+d),\qquad(x+a+c,\,r-b+d).
\end{equation}
Its rows are $r-b$ to $r+d$, and its \emph{span} is $a+c$.
\end{lemma}

\begin{proof}
No side is vertical. Two corners in the leftmost column $x$ would therefore
be opposite, forcing the other two into column $x$ as well (their mean column
is $x$). The same argument applies to rows. Both sides from $\ell$ thus go
right, one up and one down, and perpendicularity gives $ac=bd$.
\end{proof}

\begin{lemma}[span bound]\label{lem:span}
Every tilted rectangle of the $h\times m$ grid has span $a+c\le W_h$.
\end{lemma}

\begin{proof}
Here $b+d\le h-1$, and $(a-1)(c-1)\ge0$ gives $a+c\le bd+1\le W_h$, since $bd$ is an integer at most $(b+d)^2/4$.
\end{proof}

So $W_h$ is \seqnum{A033638}$(h-1)$, attained for $h\ge3$ and $m>W_h$ by the sides $(1,-b)$ and $(bd,d)$ with $b+d=h-1$ and $|b-d|\le1$. The \emph{height} and \emph{width} of a rectangle are the ranges of the rows and columns of its corners.

The \emph{diamond} $D(i,j)$ is the square with corners $(i,j+1)$, $(i+1,j)$, $(i+2,j+1)$, and $(i+1,j+2)$. The rectangles $\kA_i$, $\kB_i$, $\kC_i$, and $\kE_i$ have one corner in each of the columns $i$ to $i+3$, in the rows $3412$, $3142$, $2413$, and $2143$ from left to right.

\begin{lemma}[tilted rectangles on narrow grids]\label{lem:tilted-narrow}
The tilted rectangles of the $3\times m$ grid are the diamonds $D(i,1)$. Those of the $4\times m$ grid are $D(i,1)$, $D(i,2)$, $\kA_i$, $\kB_i$, $\kC_i$, and $\kE_i$.
\end{lemma}

\begin{proof}
Each listed set is a rectangle. On three rows $b=d=1$, so $a=c=1$ and $\ell$ is in row $2$. On four rows either $b=d=1$, which gives the diamonds, or $\{b,d\}=\{1,2\}=\{a,c\}$ and $\ell$ is in row $1+b$. The sides $(2,-2),(1,1)$; $(1,-2),(2,1)$; $(2,-1),(1,2)$; and $(1,-1),(2,2)$ give $\kA$, $\kB$, $\kC$, and $\kE$.
\end{proof}

\begin{lemma}[boundary cross]\label{lem:cross}
The first row and the first column of the $h\times m$ grid form a valid blackout, so $f(h,m)\ge h+m-1$.
\end{lemma}

\begin{proof}
By Lemma~\ref{lem:threecorners}, we need only check presentations with at most two points. An axis-aligned rectangle $\{x_1,x_2\}\times\{y_1,y_2\}$ shows $X\times Y$ with $X=\{x_1,x_2\}\setminus\{1\}$ and $Y=\{y_1,y_2\}\setminus\{1\}$, which determine it, since $|X|=1$ exactly when $1\in\{x_1,x_2\}$, and likewise for $Y$. By Lemma~\ref{lem:leftmost} a tilted rectangle has at most two black corners, $\ell$ and $(x+a,r-b)$. If both are black, then $\ell=(1,1+b)$ and it shows $p=(1+c,1+b+d)$ and $q=(1+a+c,1+d)$. These determine $a,b,c,d$ and differ in both coordinates, so no axis-aligned rectangle shows $\{p,q\}$.
\end{proof}

\begin{lemma}[axis-aligned rivals]\label{lem:rival}
Let no axis-aligned rectangle be fully black. A \emph{rival} of a set $P$ of at most two white points is an axis-aligned rectangle that shows exactly $P$. The empty set has no rival, and $P$ has one exactly when it is a singleton $\{(x,r)\}$ with $\{r,s\}\in F$ for some $s\in B_x$, a vertical pair $\{(x,r),(x,s)\}$ with $\{r,s\}\in F$, a horizontal pair $\{(x,r),(z,r)\}$ with $B_x\cap B_z\ne\emptyset$, or an oblique pair $\{(x,r),(z,s)\}$ with $x\ne z$, $r\ne s$, and $(x,s),(z,r)\in S$.
\end{lemma}

\begin{proof}
A rival of $P\ne\emptyset$ has rows $\{r,s\}$, columns $\{x,z\}$, and its other corners black. As $(x,r)$ is white, this is the listed condition, which forces $z\ne x$ and $s\ne r$.
\end{proof}
