\section{Four rows}\label{sec:four}

\begin{theorem}[four rows]\label{thm:fourvalue}
For every $m\ge6$ we have $f(4,m)=m+4$.
\end{theorem}

Theorem~\ref{thm:rowbound} gives the upper bound. For the lower bound take
\begin{equation}\label{eq:fourS}
S=\{(i,1):2\le i\le m-1\}\cup\{(m-1,2),(m,2),(1,3),(2,3),(1,4),(m,4)\}.
\end{equation}
Here $|S|=m+4$ and $G_S$ is the eight-cycle $r_1\,c_2\,r_3\,c_1\,r_4\,c_m\,r_2\,c_{m-1}$ with leaves $c_3,\ldots,c_{m-2}$ on $r_1$.

\begin{proof}[Proof of Theorem~\ref{thm:fourvalue}]
As $G_S$ has no four- or six-cycle, axis-aligned rectangles have distinct presentations and none is hidden, by Lemma~\ref{lem:girth}.

By Lemma~\ref{lem:tilted-narrow}, tilted corners are black in row $1$ and white in row $4$. The other black corners are $(i+2,2)$ if $i\ge m-3$ and $(i+1,3)$ if $i=1$ in $D(i,1)$, $(i,3)$ if $i\le2$ and $(i+1,2)$ if $i=m-2\ge4$ in $D(i,2)$, and $(i,3)$ if $i\le2$ and $(i+3,2)$ if $i\ge m-4$ in $\kA_i$ and $\kB_i$. By Lemma~\ref{lem:threecorners} a rectangle with at most one black corner is safe, leaving those below, where $s=1$ for $\kA_i$ and $s=2$ for $\kB_i$.
\begin{center}\small
\begin{tabular}{@{}lll@{}}
\toprule
rectangle & which $i$ & corners shown\\
\midrule
$D(1,1)$ & & $(1,2),(3,2)$\\
$D(i,1)$ & $i\in\{m-3,m-2\}$ & $(i,2),(i+1,3)$\\
$\kA_i$ or $\kB_i$ & $i\in\{1,2\}$, $i\notin\{m-4,m-3\}$ & $(i+s,4),(i+3,2)$\\
$\kA_i$ or $\kB_i$ & $i\in\{m-4,m-3\}$, $i\notin\{1,2\}$ & $(i,3),(i+s,4)$\\
$\kA_i$ or $\kB_i$ & $i=2$, $m=6$ & $(i+s,4)$\\
\bottomrule
\end{tabular}
\end{center}

The first four lines show different row sets. Within a line, the point in row
$2$ or $3$ fixes $i$, and the point in row $4$ fixes $s$.

No row $r$ has $(1,r)$ and $(3,r)$ black, so $\{(1,2),(3,2)\}$ has no rival. In the next three lines the axis-aligned rectangle through the shown points has white corner $(i,3)$, $(i+s,2)$, or $(i,4)$, respectively, since $i\ge3$, $i+s\le4<m-1$, and $3\le i\le m-3$ there. In the last line $m=6$, and the shown point $(z,4)$ has $z\in\{3,4\}$ and no rival, as column $z$ is black only in row $1$ and no column is black in rows $1$ and $4$.
\end{proof}

By Theorem~\ref{thm:equality}, for $m\ge4$ the axis-valid blackouts with $m+4$ points are the $3\,(m)_4\,4^{m-4}$ \emph{candidates}, with four doubles whose row pairs form a four-cycle $F$ and singletons elsewhere.

\begin{corollary}[widths four and five, computer-assisted]\label{cor:fourboundary}
We have $f(4,4)=7$ and $f(4,5)=8$.
\end{corollary}

\begin{proof}
Lemma~\ref{lem:cross} gives the lower bounds, and a computer finds a repeated presentation in all $72$ and $1{,}440$ candidates.
\end{proof}

\begin{theorem}[no hidden rectangle]\label{thm:hidden}
For $m\ge4$, no valid blackout of the $4\times m$ grid hides a rectangle.
\end{theorem}

\begin{proof}
By Lemma~\ref{lem:diff} it suffices to write each corner set as $C(R_1)\triangle C(R_2)$ with $R_1\ne R_2$. For an axis-aligned rectangle, keep its rows and pair each of its columns $x,y$ with a third column $w$. For $1\le i\le m-3$ let $\mathsf Q_i=\{(i,2),(i,3),(i+3,2),(i+3,3)\}$. Then
\begin{equation}\label{eq:fourxor}
\begin{gathered}
\kC_i=\kA_i\,\triangle\,\mathsf Q_i,\qquad \kE_i=\kB_i\,\triangle\,\mathsf Q_i,\\
D(i+1,1)=D(i,2)\,\triangle\,\kA_i,\qquad D(i+1,2)=D(i,1)\,\triangle\,\kE_i,
\end{gathered}
\end{equation}
Each identity can be solved for any of its three rectangles. The first line covers $\kA_i$, $\kB_i$, $\kC_i$, and $\kE_i$, and the second, with $i=j-1$ or $i=j$, covers $D(j,1)$ and $D(j,2)$, since $j\ge2$ or $j\le m-3$ as $m\ge4$.
\end{proof}

\begin{theorem}[forbidden-block bound]\label{thm:foursurvival}
For every $m\ge6$ we have $a_4(m)\le3\,(m)_4\,4^{m-4}(53/64)^k$, where
$k=\max(0,\lfloor m/4\rfloor-4)$. So $a_4(m)=O\bigl(m^4\,212^{m/4}\bigr)$.
\end{theorem}

\begin{proof}
Optima are candidates, with $3\,(m)_4$ choices of $F$, the double columns, and their
pairs. A \emph{block} is a run of four singleton columns, whose word lists
their black rows. For a block from column $x$, the \emph{template words} $2413$,
$3412$, $2143$, and $3142$ are the corner rows of $\kC_x$, $\kA_x$, $\kE_x$, and $\kB_x$. Let
$T$ be one, with word $t_1t_2t_3t_4$.

A \emph{swap word} exchanges two letters $t_p$ and $t_q$. Such a block is impossible.
The rectangle $T$ and the axis-aligned rectangle through $(x+p-1,t_p)$ and
$(x+q-1,t_q)$ would show exactly these two points. Template words are the permutations with both end letters in $\{2,3\}$.
Swapping two end or two middle letters gives one again, and swapping an end
and a middle letter gives the sixteen with one end letter in $\{2,3\}$, so the $20$ swap words are these and the four template words.

An \emph{edit word} replaces a letter $t_p=r$ by $s$ with $\{r,s\}\in F$. No block carries
one, since $T$ and the rectangle on rows $r,s$ and columns $x+p-1$ and $d$, the double column
of $\{r,s\}$, would both show only $(x+p-1,r)$. Each template word has $8$ edit words.
Two template words share one only if they differ in two positions holding $r,s$ in opposite
orders, and then share two if $\{r,s\}\in F$, which happens for two pairs with
$\{r,s\}=\{2,3\}$ and two with $\{1,4\}$. So there are at least $32-8=24$ edit words. They repeat a letter, so none is a swap word, and at most
$256-20-24=212=4^4\cdot53/64$ words are allowed in a block.

At most four of the $\lfloor m/4\rfloor$ column groups $4j-3,\ldots,4j$ contain a double
column, so at least $k$ are blocks. Other columns have at most $4$ values.
\end{proof}

\begin{lemma}[six columns of memory]\label{lem:fourmemory}
Let $m\ge6$, and let $F$ be the edge set of a four-cycle on the rows. A word over the
four singleton letters and the pairs of $F$, each pair used once, represents an optimum
if and only if (i) no tilted rectangle is hidden, (ii) no tilted rectangle shares its
presentation with an axis-aligned rectangle whose columns are among its own and the
double columns, and (iii) no two distinct tilted rectangles in a block of at most seven
consecutive columns share a presentation.
\end{lemma}

\begin{proof}
An optimum satisfies all three conditions, with (i) following from
Theorem~\ref{thm:hidden}. Conversely, a word of the stated form is axis-valid by
Theorem~\ref{thm:equality}. It hides no axis-aligned rectangle, since that would give a
four-cycle in $G_S$, and (i) excludes hidden tilted rectangles.

Suppose two distinct rectangles have the same presentation $P$. Then $P\ne\emptyset$,
and at least one rectangle is tilted. If the other is axis-aligned, each of its columns
either meets $P$ or has two black corners. A column meeting $P$ belongs to the tilted
rectangle, and a column with two black corners is a double column. Condition (ii)
therefore excludes this pair. If both rectangles are tilted, they share a column $c$
meeting $P$. Each occupies at most four consecutive columns, so together they lie in
columns $c-3$ to $c+3$, contrary to (iii). The word is thus valid and, with $m+4$ black
points, is an optimum.
\end{proof}

A tilted rectangle of a word over the four singleton letters and the pairs of $F$ \emph{fails
locally} if it is hidden, if it shows a set with a rival by Lemma~\ref{lem:rival} read with pair
set $F$, or if another tilted rectangle of the word shows the same set.

\begin{proposition}[the automaton]\label{prop:fourautomaton}
Let $F$ be the edge set of a four-cycle on the rows. The automaton $\mathcal A_F$ reads
the four singleton letters and the pairs in $F$. It starts at $(\epsilon,\emptyset)$,
where $\epsilon$ is the empty word. A state $(u,U)$ records the suffix of at most six
letters read so far and the set $U\subseteq F$ of pairs already used.

On reading $y$, it rejects if $y\in U$ or if a tilted rectangle in $uy$ ending in the
new column fails locally. Otherwise it replaces $u$ by the suffix of $uy$ of length
$\min(6,|uy|)$ and $U$ by $U\cup(\{y\}\cap F)$. A word is accepted if it is read to the end
without rejection and the final register is $U=F$. For each $m\ge6$, the accepted words
of length $m$ are exactly the optima of the $4\times m$ grid with pair set $F$.
\end{proposition}

\begin{proof}
Fix $m\ge6$. Tilted rectangles occupy at most four consecutive columns, so every such
rectangle is contained in $uy$ when its last column is read. An accepted word uses each
pair of $F$ exactly once and is a candidate. Presentations with at least three points
are unique by Lemma~\ref{lem:threecorners}. The local tests exclude hidden tilted
rectangles and axis-aligned rivals, giving (i) and (ii) of Lemma~\ref{lem:fourmemory}.
For (iii), any two tilted rectangles in at most seven consecutive columns are both in the
window when the later of their last columns is read, so the test compares their
presentations. The lemma makes the word an optimum.

Conversely, an optimum of width $m$ with pair set $F$ uses each pair once. It hides no
rectangle by Theorem~\ref{thm:hidden}, and no two rectangles have the same presentation.
Lemma~\ref{lem:rival}, applied with its pair set $F$, therefore finds no rival at any
step. None of the local tests rejects the word, and its final register is $F$.
\end{proof}

\begin{theorem}[the four-row generating function, computer-assisted]\label{thm:fourgfexact}
There are coprime integer polynomials $P_4=26z^6-826z^7+\cdots$ of degree $444$ and $Q_4=\sum_{j=0}^{420}q_jz^j$ of degree $420$ with $q_0=1$ and $\sum_{m\ge6}a_4(m)z^m=P_4/Q_4$. The recurrence $\sum_jq_j\,a_4(n-j)=0$ holds for every $n\ge445$ and fails at $n=444$. Over the integers
\[
Q_4=(1-z)^2(1+z)^2\,g_5^3\,\tilde g_5^3\,g_6^3\,g_{11}^5\,g_{17}^3\,g_{19}^3\,g_{20}^5\,g_{21}^5,
\]
where $g_d$ and $\tilde g_d$ are irreducible of degree $d$ with constant term $1$.
\end{theorem}

\begin{proof}
Proposition~\ref{prop:fourautomaton} identifies the counted words with the optima. Berlekamp--Massey modulo eight primes on $a_4(m)$, $m\le1500$, gives $P_4$ and $Q_4$. The reachable states give a transfer matrix $T_F$ for each $F$, and its state vectors $v(n)=v(0)T_F^n$ satisfy $\sum_jq_jv(n_F-j)=0$ at $n_F=436$, $435$, and $445$, so powers of $T_F$ give the recurrence for $n\ge445$, and $\deg P_4=444$ forces its failure at $n=444$. SymPy factors $Q_4$.
\end{proof}

\begin{corollary}[growth, computer-assisted]\label{cor:fourgrowth}
Let $\rho_4=3.351996777310\ldots$ be the largest real root of $p(x)=x^{20}g_{20}(1/x)=x^{20}-4x^{19}+2x^{18}+\cdots-26x^2-4x+8$. There is a polynomial $B$ of degree $4$ with $a_4(m)=\rho_4^m B(m)+O((1000/341)^m)$, so $a_4(m)\sim c\,m^4\rho_4^m$ with $c\approx8.1611\times10^{-5}$.
\end{corollary}

\begin{proof}
Exact root isolation on the factors of $Q_4$ shows that $r=1/\rho_4=0.29832\ldots$ is its only root of modulus at most $0.341$, simple in $g_{20}$ and so of order five. So $P_4/Q_4=\sum_{j=1}^{5}c_j(1-z/r)^{-j}+E(z)$, where $E$ has coefficients $O((1000/341)^m)$ and, by coprimality, $c_5=H(r)\ne0$ for $H(z)=P_4(z)(1-z/r)^5/Q_4(z)$. As $[z^m](1-z/r)^{-j}=\rho_4^m\binom{m+j-1}{j-1}$, we get $B(m)=\sum_jc_j\binom{m+j-1}{j-1}$ and $c=c_5/24$. Interval arithmetic puts $c$ between $8.1610970775\times10^{-5}$ and $8.1610970776\times10^{-5}$.
\end{proof}
