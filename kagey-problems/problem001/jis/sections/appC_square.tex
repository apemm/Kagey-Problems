\section{Square grids}\label{app:square}

Let $n\ge2$, $L=1+\ln n$, $\mathcal R$ the set of rectangles of the $n\times n$ grid, tilted ones included, and $d_v$ the number with corner $v$.

\begin{lemma}\label{lem:compress}
If rectangles $R_1\ne R_2$ have exactly two common corners $p$ and $q$, and $\{p,q\}$ is a side of both or a diagonal of both, then $C(R_1)\triangle C(R_2)$ is the corner set of a rectangle.
\end{lemma}

\begin{proof}
For a common side, the other corners are $p+u,q+u$ and $p+v,q+v$ with $u\ne v$ nonzero and perpendicular to $q-p$, so they are distinct and form a rectangle with sides $q-p$ and $v-u$. For a common diagonal, the other corners are $o\pm u$ and $o\pm v$ with $o=(p+q)/2$, $|u|=|v|$, and $v\ne\pm u$. The segments $[o-u,o+u]$ and $[o-v,o+v]$ share midpoint and length, so they are diagonals of a rectangle.
\end{proof}

Let $r_2(t)$ count the integer pairs $(a,b)$ with $a^2+b^2=t$, and $\tau(t)$ the divisors of $t$. By Hardy and Wright \cite[Sections~16.9, 16.10, and~18.1]{hardywright2008}, $r_2(t)\le4\tau(t)$, and each $\varepsilon>0$ has a $K_\varepsilon$ with $\tau(t)\le K_\varepsilon t^\varepsilon$ for $t\ge1$.

Let $\mathcal A$ consist of the sets $C(R)$ with $R\in\mathcal R$ and the sets $C(R_1)\triangle C(R_2)$ whose exactly two common corners form a side of one and a diagonal of the other, and $\mathcal B$ of the sets $C(R_1)\triangle C(R_2)$ with exactly one common corner.

\begin{lemma}\label{lem:counts}
\begin{enumerate}[(a)]
\item Each point $v$ has $d_v\le8n^2L$, and $|\mathcal R|\le2n^4L$.
\item For every $\varepsilon>0$, $|\mathcal A|\le A_\varepsilon n^{4+2\varepsilon}L$ with $A_\varepsilon=2(1+16\cdot2^\varepsilon K_\varepsilon)$, and $|\mathcal B|\le32n^6L^2$.
\item A blackout that contains no member of $\mathcal A\cup\mathcal B$ is valid.
\end{enumerate}
\end{lemma}

\begin{proof}
(a) The sides at $v$ are $k(a,b)$ and $l(-b,a)$ with $\gcd(a,b)=1$, $k\ge1$, and $l\ne0$. With $d=\max(|a|,|b|)$ there are $8d$ choices of $(a,b)$, at most $(n-1)/d$ of $k$, and $2(n-1)/d$ of $l$. Each rectangle counts twice, so $2d_v\le\sum_{d=1}^{n-1}16(n-1)^2/d\le16n^2L$, and $4|\mathcal R|=\sum_vd_v\le8n^4L$.

(b) A rectangle with diagonal $\{p,q\}$ is fixed by its other diagonal, an integer vector of squared length $t=|p-q|^2<2n^2$, so each side of each $R\in\mathcal R$ is a diagonal of at most $r_2(t)\le4K_\varepsilon(2n^2)^\varepsilon$ rectangles. Hence $|\mathcal A|\le|\mathcal R|(1+16K_\varepsilon2^\varepsilon n^{2\varepsilon})\le A_\varepsilon n^{4+2\varepsilon}L$. Members of $\mathcal B$ come from two rectangles at a corner $v$, so $|\mathcal B|\le\sum_v\binom{d_v}2\le\frac{n^2}2(8n^2L)^2$.

(c) By Lemma~\ref{lem:diff}, an invalid blackout $S$ contains $C(R_1)\triangle C(R_2)$ for some $R_1\ne R_2$, which share at most two corners by Lemma~\ref{lem:threecorners}. With two, this set lies in $\mathcal A$ by Lemma~\ref{lem:compress} or by definition; with one, in $\mathcal B$; with none, $C(R_1)\subseteq S$ lies in $\mathcal A$.
\end{proof}

\begin{theorem}[deletion method]\label{thm:alteration}
We have
\[
\liminf_{n\to\infty}\frac{f(n,n)\,(1+\ln n)^{2/5}}{n^{6/5}}\ge\frac5{6\cdot192^{1/5}}=0.2911\ldots.
\]
In particular $f(n,n)-2n\to\infty$.
\end{theorem}

\begin{proof}
Following Alon and Spencer~\cite[Section~3.2]{alonspencer2016}, put $a=192^{-1/5}$, $T=n^{6/5}L^{-2/5}$, and $\theta=a\,n^{-4/5}L^{-2/5}\in(0,1)$, and select grid points independently with probability $\theta$. Let $X$, $Y$, and $Z$ count the selected points and the fully selected members of $\mathcal A$ and $\mathcal B$. Then
\[
\mathbb E(X-Y-Z)=n^2\theta-|\mathcal A|\theta^4-|\mathcal B|\theta^6 .
\]
Here $n^2\theta=aT$, and Lemma~\ref{lem:counts}(b) with $\varepsilon=1/10$ gives $|\mathcal B|\theta^6\le32a^6T$ and $|\mathcal A|\theta^4\le A_{1/10}a^4n^{-1/5}T$. Since $32a^5=\frac16$, some selection $S_0$ has $X-Y-Z\ge(\frac56a-A_{1/10}a^4n^{-1/5})T$. Deleting a point of each member of $\mathcal A\cup\mathcal B$ inside $S_0$ leaves at least $X-Y-Z$ points, valid by Lemma~\ref{lem:counts}(c). So $f(n,n)/T\ge\frac56a-A_{1/10}a^4n^{-1/5}\to5/(6\cdot192^{1/5})$, and $T/n\to\infty$ gives $f(n,n)-2n\to\infty$.
\end{proof}
