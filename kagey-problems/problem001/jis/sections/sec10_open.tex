\section{Open questions}\label{sec:open}

\begin{enumerate}[(1),leftmargin=*]
\item \emph{Five-row counts.} The series $\sum_{m\ge0}a_5(m)z^m$ is rational by
Theorem~\ref{thm:periodic}. Find it, and the growth rate of $a_5(m)$ (Remark~\ref{rem:fivecounts}).
\item \emph{Six rows below width $15$.} The first stored blackout in
\nolinkurl{jis/data/small_cells_sat_6x9_b16.json} passes the
test of Theorem~\ref{thm:collar} with seven columns black only in row $1$ added on each side
(\nolinkurl{six_row_collar_6x9_check.py}). With Theorem~\ref{thm:sixrow} this gives
$m+7\le f(6,m)\le m+8$ for $9\le m\le14$. What are these values?
\item \emph{Seven and more rows.} Is the row bound $m+q_h$ attained at every large width for
every $h\ge7$? By Theorem~\ref{thm:collar}, one valid $7\times m$ blackout of $m+12$ points
that passes the collar test would settle $h=7$.
\item \emph{Squares.} For $n\ge3$, Lemma~\ref{lem:girth} and Neuwirth's bound for
bipartite graphs of girth eight~\cite[Corollary~16]{neuwirth2001} give the known bound
$f(n,n)<n^{4/3}+\frac23n-\frac29n^{2/3}-\frac{20}{81}n^{1/3}$. Theorem~\ref{thm:alteration}
gives $f(n,n)\ge0.29\,n^{6/5}(1+\ln n)^{-2/5}$ for large $n$, and
Lavrov~\cite{kageyMSE} used the same method for the triangular version.
What is the true order, and the least $n$ with $f(n,n)>2n$?
\item \emph{Proofs by hand.} Is there a proof without a computer that the $5\times5$ optimum
is unique up to symmetry, or that the four-row generating function has a dominant pole of
order five?
\end{enumerate}
