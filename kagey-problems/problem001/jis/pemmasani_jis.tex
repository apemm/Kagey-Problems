\begin{abstract}
Kagey's Problem 001 concerns a grid of points, some of them blacked out. A rectangle
with its corners on grid points, tilted or not, shows only its white corners. We ask how
many points can be blacked out while every rectangle stays determined by what it shows.
We study the maximum blackout and the number of optima, tabulated in A397315 and A399580
of the On-Line Encyclopedia of Integer Sequences. A row-column graph and Mantel's theorem
bound the maximum by the width plus a quarter of the squared height. We find the maximum
on up to five rows at every width, and on six rows from width 15 on. The five-row result
refutes our earlier guess that the maximum is the sum of the two sides. We count the three-row optima exactly and
find the growth rate on four rows. At every fixed height, the maximum minus the width is
eventually periodic, and the optimum counts have a rational generating function. Some
values rest on computer checks.
\end{abstract}

Let $f(h,m)$ be the largest size of a valid blackout, and
$a_h(m)$ the number of these. Our OEIS entries \seqnum{A397315} and
\seqnum{A399580}~\cite{oeis} first recorded the conjecture $f(3,m)=m+2$, which the Omega
Institute~\cite{trureturing2026} proved in Lean, and the guess $f(h,m)=h+m$ for
$\min(h,m)\ge4$ apart from $f(4,4)=7$ and $f(4,5)=8$. The guess first fails at
$5\times8$ (Theorem~\ref{thm:fiverow}). Put $q_h=\lfloor h^2/4\rfloor$. Results that rest on
a computer check are marked computer-assisted in their statements.

\begin{introthm}[narrow grids]\label{thm:A}
\leavevmode
\begin{enumerate}[(a)]
\item We have $f(h,m)\ge h+m-1$, and $f(h,m)\le m+q_h$ if $m\ge3$.
\item We have $f(2,m)=m+1$ and $f(3,m)=m+2$ for $m\ge3$, $f(4,m)=m+4$ for $m\ge6$,
and $f(5,5)=10$.
\item (Computer-assisted, with checked certificates for $f(5,6)$ and $f(5,7)$, and the last
claim checked in Lean.) We have $f(4,4)=7$, $f(4,5)=8$,
$f(5,6)=11$, $f(5,7)=12$, and $f(5,m)=m+6$ for $m\ge8$.
\item (Computer-assisted for $m\ge9$, with checked certificates for $m\le14$ and checked in Lean
for $m\ge15$.) We have $f(6,m)\le m+8$ for $8\le m\le14$, and $f(6,m)=m+9$ for $m\ge15$.
\item (Computer-assisted, with checked certificates.) We have $f(6,6)=12$,
$f(6,7)=13$, $f(7,7)=14$, and $f(6,8)=15$.
\end{enumerate}
\end{introthm}

Part (a) is Lemma~\ref{lem:cross} and Theorem~\ref{thm:rowbound}, (b) is
Theorem~\ref{thm:strip}, Section~\ref{sec:three}, and Theorems~\ref{thm:fourvalue}
and~\ref{thm:fivefive}, (c) is Corollary~\ref{cor:fourboundary} and
Theorem~\ref{thm:fiverow}, (d) is Theorem~\ref{thm:sixrow}, and (e) is
Theorem~\ref{thm:refute}. For (c) and (d) the finite collar theorem
(Theorem~\ref{thm:collar}) reduces validity at every larger width to one finite board.

\begin{introthm}[three-row optima]\label{thm:B}
For $m\ge4$ we have $a_3(m)=4(m-1)(3m+2)\,3^{m-5}+E_m$, where $E_m>0$ counts the optima
with a column black in rows $1$ and $3$, and $E_m=O(m^2(1+\sqrt3)^m)$. So
$a_3(m)\sim\frac4{81}m^23^m$, and a fraction tending to $4/27$ of the $m^23^{m-1}$
axis-valid candidates is valid.
The series $\sum_{m\ge3}a_3(m)z^m$ is rational. (Computer-assisted.) Its reduced
denominator is $(1-3z)^3(1-2z-2z^2)^3(1+2z^2)^3$, and
$E_m\sim\frac{19+11\sqrt3}{432}m^2(1+\sqrt3)^m$.
\end{introthm}

See Theorems~\ref{thm:threecount}, \ref{thm:threeclosed}, and~\ref{thm:threegf}, and
Section~\ref{app:threegf}.

\begin{introthm}[four rows]\label{thm:C}
If $m\ge4$, no valid blackout of the $4\times m$ grid hides all corners of a
rectangle. The series $\sum_{m\ge6}a_4(m)z^m$ is rational with
reduced denominator of degree $420$, and $a_4(m)\sim c\,m^4\rho_4^m$ with
$c\approx8.1611\times10^{-5}$ and $\rho_4=3.35199\ldots$, an algebraic number of degree
$20$. (Computer-assisted.)
\end{introthm}

See Theorems~\ref{thm:hidden} and~\ref{thm:fourgfexact} and Corollary~\ref{cor:fourgrowth}.

\begin{introthm}[every fixed height]\label{thm:E}
Fix $h\ge1$ and write a blackout as the word $B_1\cdots B_m$ of the black rows of its
columns. The valid words form a regular language. If $h\ge2$, then $f(h,m)-m$ is
eventually periodic in $m$, and $\sum_{m\ge0}a_h(m)z^m$ is rational.
\end{introthm}

See Theorems~\ref{thm:regular} and~\ref{thm:periodic}.

\paragraph{Prior work.} The game asks for a test
cover~\cite{bellitto2018,foucaud2015,katona1966,moretshapiro1985}
(Corollary~\ref{cor:hitting}). Forbidding black axis-aligned rectangles is the
Zarankiewicz problem~\cite{reiman1958}. Distinguishing them forbids four- and six-cycles
in the row-column graph (Lemma~\ref{lem:girth}), and de Caen and
Sz\'ekely~\cite{decaenszekely1992,decaenszekely1997} study such graphs with unbalanced parts.
Neuwirth~\cite[Theorem~1(ii) and Proposition~17]{neuwirth2001} bounds their edges, so the row
bound is known in graph form. Our proof with Mantel's theorem~\cite{mantel1907} also finds and
counts the equality case. None of these works sees tilted rectangles. Column transfer~\cite{alanko2011,euler2005,mertens2024},
\cite[Proposition~V.6]{flajolet2009} handles fixed heights, and Theorem~1(3) of Guilhot,
Little, and Parkinson~\cite{glp2025} with Theorem~\ref{thm:regular} also gives
Theorem~\ref{thm:E}.

\section{Conventions}\label{sec:conv}

\begin{enumerate}[(1),leftmargin=*]
\item An $h\times m$ grid has $h$ rows, $m$ columns, and point set $V$. A $2\times m$ grid
is a \emph{strip}. The point $(i,j)$ lies in column $i$ and row $j$, with $(1,1)$ at the
top left (Figure~\ref{fig:game}(a)).
\item A \emph{rectangle} $R$ is its set $C(R)$ of four corners, grid points spanning a
rectangle of positive area. It is \emph{axis-aligned} if its sides are parallel to the
grid lines, and \emph{tilted} otherwise.
\item A \emph{blackout} is a set $S\subseteq V$ with \emph{kept set} $K=V\setminus S$. The
\emph{presentation} of $R$ is $\pres(R)=C(R)\setminus S$.
\item Column $i$ has \emph{black rows} $B_i=\{j:(i,j)\in S\}$, and is \emph{empty}, a
\emph{singleton}, a \emph{double}, or a \emph{triple} when $|B_i|$ is $0$, $1$, $2$, or
$3$, and \emph{full} when $|B_i|=h$. The \emph{pair set} $F$ is the set of row pairs
contained in some $B_i$, and $q_h=\lfloor h^2/4\rfloor$ is \seqnum{A002620}$(h)$.
\end{enumerate}

\begin{definition}[valid blackout]\label{def:valid}
A blackout $S$ is \emph{valid} if $\pres(R_1)\ne\pres(R_2)$ whenever $R_1\ne R_2$. The
\emph{maximum blackout} $f(h,m)$ is the largest size of a valid blackout of the
$h\times m$ grid, and $a_h(m)$ counts the valid blackouts of that size, with their
coordinates.
\end{definition}

By symmetry $f(h,m)=f(m,h)$ and $a_h(m)=a_m(h)$. Width $0$ and height $1$ have no
rectangles, so $f(h,0)=0$, $a_h(0)=1$, $f(1,m)=m$, and $a_1(m)=1$. An empty presentation
is allowed, so a valid blackout hides at most one rectangle.

\begin{lemma}[difference sets]\label{lem:diff}
For sets $A$, $B$, and $S$, we have $A\setminus S=B\setminus S$ if and only if
$A\triangle B\subseteq S$.
\end{lemma}

\begin{proof}
Both conditions say that no point outside $S$ lies in exactly one of $A$ and $B$.
\end{proof}

\begin{corollary}[hitting sets]\label{cor:hitting}
A blackout $S$ is valid if and only if $K$ meets $C(R_1)\triangle C(R_2)$ for all distinct
rectangles $R_1$ and $R_2$.
\end{corollary}

\begin{proof}
Apply Lemma~\ref{lem:diff} to $C(R_1)$ and $C(R_2)$.
\end{proof}

So subsets of valid blackouts are valid, and showing that no blackout of $k$ points is
valid proves $f(h,m)<k$.

\begin{lemma}[three corners suffice]\label{lem:threecorners}
Three corners of a rectangle determine it, so distinct rectangles with the same
presentation show at most two corners.
\end{lemma}

\begin{proof}
Three corners form a right triangle, whose only right angle is at the corner $q$ adjacent
to the other two, $p$ and $r$. So the fourth corner is $p+r-q$.
\end{proof}
\section{Strips}\label{sec:strip}

On two rows every rectangle is axis-aligned, since a tilted one meets three rows: the
neighbors of a lowest corner lie strictly above it, as no side is horizontal, and the opposite corner lies above them.

\begin{theorem}[strip theorem]\label{thm:strip}
Let $m\ge3$.
\begin{enumerate}[(a)]
\item A blackout of the $2\times m$ strip is valid if and only if at most one of its columns is full.
\item We have $f(2,m)=m+1$, and the maximum blackouts are those with one full column
and one black point in every other column.
\item There are $a_2(m)=m\cdot2^{m-1}$ maximum blackouts, so row $2$ of
\seqnum{A399580} is \seqnum{A001787} from $m=3$ on.
\end{enumerate}
For $m=2$ there is one rectangle, so $f(2,2)=4$ and $a_2(2)=1$.
\end{theorem}

\begin{proof}
(a) Distinct rectangles share at most one column, so their difference set contains all
four points of two columns $p$ and $q$. Every pair arises, since $m\ge3$ gives a third column $z$ and the rectangles on
$\{z,p\}$ and $\{z,q\}$. By Corollary~\ref{cor:hitting}, $S$ is
valid if and only if $K$ meets every two columns, that is, at most one column is full.

(b, c) By (a), $|K|\ge m-1$, so $|S|\le m+1$, with equality exactly for the blackouts
in (b). There are $m$ choices of full column and two in each other column.
\end{proof}

\begin{proposition}[orbits and kings]\label{prop:kings}
For $m\ge3$ the number $o_m$ of maximum blackouts of the $2\times m$ strip up to the row
reflection and the column reversal is $m\,2^{m-3}+[m\text{ odd}]\,2^{(m-3)/2}$. It equals
\seqnum{A322284}$(m-1)$, which counts $m-1$ nonattacking kings on a $2\times(2m-2)$
board up to symmetry.
\end{proposition}

\begin{proof}
By Burnside's lemma, $4o_m$ is $a_2(m)$ plus the optima fixed by the reflection (none)
and by the reversal and the half turn (each fixes none for even $m$, and $2^{(m-1)/2}$
with a central full column for odd $m$).

Put $n=m-1$. Nonattacking kings on the $2\times2n$ board are columns $c_1<\cdots<c_n$ with
$c_{j+1}\ge c_j+2$, and any rows $r_j$. The gaps $c_1-1$, $c_{j+1}-c_j-2$, and $2n-c_n$
are nonnegative and sum to $1$, so exactly one, say the $e$th, is $1$, and
$c_j=2j-1+[j\ge e]$. Sending this to the optimum with full column $e$ and other black
points in rows $r_1,\ldots,r_n$ from left to right is a bijection. It commutes with the
reflection, which swaps the rows on both sides, and with the reversal, which sends $e$ to
$n+2-e$ and $c$ to $2n+1-c$ and reverses the rows, since $2n+1-c_{n+1-j}=2j-1+[j\ge n+2-e]$.
So the orbits correspond.
\end{proof}
\section{The row-column graph and general bounds}\label{sec:graph}

A blackout $S$ is \emph{axis-valid} if no two axis-aligned rectangles have the same
presentation. A rectangle showing at most two corners is \emph{weak}. The \emph{row-column graph} $G_S$ has vertices $r_j$ for the rows and $c_i$ for the columns, and an edge $r_jc_i$ for each black point $(i,j)$.

\begin{lemma}[axis-aligned rectangles and short cycles]\label{lem:girth}
If $G_S$ has no four-cycle and no six-cycle, then $S$ is axis-valid. The converse holds for $m\ge3$.
\end{lemma}

\begin{proof}
For the converse, let $m\ge3$. A four-cycle $r_j\,c_i\,r_{j'}\,c_{i'}$ and a third column $w$ make $\{i,w\}\times\{j,j'\}$ and $\{i',w\}\times\{j,j'\}$ show the same points. A six-cycle $r_a\,c_x\,r_b\,c_y\,r_c\,c_z$ makes $\{x,y\}\times\{a,b\}$ and $\{y,z\}\times\{a,c\}$ black except possibly at $(y,a)$.

For the first statement, let $R_1\ne R_2$ be axis-aligned with the same presentation, so $C(R_1)\triangle C(R_2)\subseteq S$ (Lemma~\ref{lem:diff}). Opposite corners determine a rectangle, so by symmetry the common corners lie in one row $a$. With none, $R_1$ is a black four-cycle. Two, $(x,a)$ and $(y,a)$, leave the other rows $b\ne c$ and the black four-cycle $r_b\,c_x\,r_c\,c_y$. One, $(x,a)$, with $R_1=\{x,y\}\times\{a,b\}$ and $R_2=\{x,z\}\times\{a,c\}$, gives the black six-cycle $r_a\,c_y\,r_b\,c_x\,r_c\,c_z$.
\end{proof}

\begin{theorem}[row bound]\label{thm:rowbound}
Let $m\ge3$. Every axis-valid blackout of the $h\times m$ grid has at most $m+q_h$ points, and at most $h+q_m$ if also $h\ge3$. Hence $f(h,m)\le m+q_h$.
\end{theorem}

For each nonempty column $i$ choose a tree $T_i$ on $B_i$, and let $H$ be their union.

\begin{lemma}[row graph]\label{lem:H}
Let $m\ge3$ and let $S$ be axis-valid. For any choice of the trees, $H$ is simple and triangle-free, and $e(H)=|S|-t$, where $t$ is the number of nonempty columns.
\end{lemma}

\begin{proof}
By Lemma~\ref{lem:girth}, $G_S$ has no four-cycle, so $H$ has no repeated edge and $e(H)=\sum(|B_i|-1)=|S|-t$. A triangle $r_ar_br_c$ is not in one tree. Edges from two columns $i,i'$ put two of $a,b,c$ in $B_i\cap B_{i'}$, and edges from three columns give a six-cycle $r_a\,c_i\,r_b\,c_{i'}\,r_c\,c_{i''}$.
\end{proof}

\begin{lemma}[Mantel \cite{mantel1907}]\label{lem:mantel}
Let $H$ be a simple triangle-free graph on $h$ vertices, let a vertex of maximum degree $\Delta$ have neighbor set $A$, and let $B$ be the other $h-\Delta$ vertices, spanning $e(B)$ edges. Then
\[
e(H)=\sum_{b\in B}d(b)-e(B)\le(h-\Delta)\Delta-e(B)\le q_h.
\]
\end{lemma}

\begin{proof}
Every edge meets $B$, as $A$ is independent, and the degree sum over $B$ counts the edges inside $B$ twice. Also $d(b)\le\Delta$ and $(h-\Delta)\Delta\le q_h$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:rowbound}]
Lemmas~\ref{lem:H} and~\ref{lem:mantel} give $|S|=t+e(H)\le m+q_h$. Only Lemma~\ref{lem:girth} uses a third column, so for $h\ge3$ the transpose gives $|S|\le h+q_m$.
\end{proof}

\begin{theorem}[equality structure]\label{thm:equality}
Let $h\ge4$ and $m\ge3$. A blackout of the $h\times m$ grid with $m+q_h$ points is axis-valid if and only if
\begin{enumerate}[(a)]
\item every column is a singleton or a double,
\item the $q_h$ doubles have distinct row pairs, and
\item these pairs form a balanced complete bipartite graph on the rows.
\end{enumerate}
There are $p_h\,(m)_{q_h}\,h^{m-q_h}$ such blackouts, where $(m)_q$ is the falling factorial and the number $p_h$ of balanced bipartitions of the rows is $\binom h{\lfloor h/2\rfloor}$, halved for even $h$.
\end{theorem}

\begin{proof}
Let $S$ be axis-valid. Equality in Lemmas~\ref{lem:H} and~\ref{lem:mantel} forces $t=m$, $e(B)=0$, and $(h-\Delta)\Delta=q_h$ with $B$ joined to all of $A$, so $H$ is as in (c). If $|B_i|\ge3$, some $v$ has two neighbors $u,w$ in $T_i$, and as $h\ge4$ the part of $v$ has another vertex $z$. Replacing $vw$ by $uw$ in $T_i$ keeps $uz$ and $wz$ and makes the triangle $uwz$, against Lemma~\ref{lem:H}. This gives (a), and the edges of $H$, the row pairs of the doubles, give (b) and (c).

Conversely, (a) and (b) rule out four-cycles, and a six-cycle $r_a\,c_x\,r_b\,c_y\,r_c\,c_z$ would need the doubles $\{a,b\}$, $\{b,c\}$, and $\{a,c\}$, a triangle in (c), so Lemma~\ref{lem:girth} applies. The blackout determines, and is determined by, a bipartition, an injective map from its edges to columns, and a row for each other column.
\end{proof}

\begin{corollary}[small widths]\label{cor:small-m}
Let $h\ge4$ and $3\le m<q_h$. Then $f(h,m)\le m+q_h-1$. In particular $f(5,5)\le10$.
\end{corollary}

\begin{proof}
By Theorems~\ref{thm:rowbound} and~\ref{thm:equality}, a valid blackout of $m+q_h$ points has $q_h\le m$ doubles. Here $q_5=6$.
\end{proof}

Put $W_h=1+\lfloor(h-1)^2/4\rfloor$, so $W_3=2$, $W_4=3$, and $W_5=5$.

\begin{lemma}[leftmost corner]\label{lem:leftmost}
A tilted rectangle $R$ has a unique corner $\ell=(x,r)$ in its smallest column, and a unique corner in its smallest row. The sides from $\ell$ are $(a,-b)$ and $(c,d)$ with positive integers $a,b,c,d$ and $ac=bd$, and the corners are
\begin{equation}\label{eq:tilt-param}
(x,r),\qquad(x+a,\,r-b),\qquad(x+c,\,r+d),\qquad(x+a+c,\,r-b+d).
\end{equation}
Its rows are $r-b$ to $r+d$, and its \emph{span} is $a+c$.
\end{lemma}

\begin{proof}
No side is vertical, so two corners in column $x$ would be opposite, and the other two, with mean column $x$, would lie in column $x$ too. Rows are the same. So both sides from $\ell$ go right, one up and one down, and perpendicularity gives $ac=bd$.
\end{proof}

\begin{lemma}[span bound]\label{lem:span}
Every tilted rectangle of the $h\times m$ grid has span $a+c\le W_h$.
\end{lemma}

\begin{proof}
Here $b+d\le h-1$, and $(a-1)(c-1)\ge0$ gives $a+c\le bd+1\le W_h$, since $bd$ is an integer at most $(b+d)^2/4$.
\end{proof}

So $W_h$ is \seqnum{A033638}$(h-1)$, attained for $h\ge3$ and $m>W_h$ by the sides $(1,-b)$ and $(bd,d)$ with $b+d=h-1$ and $|b-d|\le1$. The same argument proves the conjecture in \seqnum{A338885} that its row $n$ has largest entry \seqnum{A033638}$(n)$.

The \emph{diamond} $D(i,j)$ is the square with corners $(i,j+1)$, $(i+1,j)$, $(i+2,j+1)$, and $(i+1,j+2)$. The rectangles $\kA_i$, $\kB_i$, $\kC_i$, and $\kE_i$ have one corner in each of the columns $i$ to $i+3$, in the rows $3412$, $3142$, $2413$, and $2143$ from left to right.

\begin{lemma}[tilted rectangles on narrow grids]\label{lem:tilted-narrow}
The tilted rectangles of the $3\times m$ grid are the diamonds $D(i,1)$. Those of the $4\times m$ grid are $D(i,1)$, $D(i,2)$, $\kA_i$, $\kB_i$, $\kC_i$, and $\kE_i$.
\end{lemma}

\begin{proof}
Each listed set is a rectangle. On three rows $b=d=1$, so $a=c=1$ and $\ell$ is in row $2$. On four rows either $b=d=1$, which gives the diamonds, or $\{b,d\}=\{1,2\}=\{a,c\}$ and $\ell$ is in row $1+b$. The sides $(2,-2),(1,1)$; $(1,-2),(2,1)$; $(2,-1),(1,2)$; and $(1,-1),(2,2)$ give $\kA$, $\kB$, $\kC$, and $\kE$.
\end{proof}

\begin{lemma}[boundary cross]\label{lem:cross}
The first row and the first column of the $h\times m$ grid form a valid blackout, so $f(h,m)\ge h+m-1$.
\end{lemma}

\begin{proof}
By Lemma~\ref{lem:threecorners} only presentations of at most two points matter. An axis-aligned rectangle $\{x_1,x_2\}\times\{y_1,y_2\}$ shows $X\times Y$ with $X=\{x_1,x_2\}\setminus\{1\}$ and $Y=\{y_1,y_2\}\setminus\{1\}$, which determine it, since $|X|=1$ exactly when $1\in\{x_1,x_2\}$, and likewise for $Y$. By Lemma~\ref{lem:leftmost} a tilted rectangle has at most two black corners, $\ell$ and $(x+a,r-b)$. If both are black, then $\ell=(1,1+b)$ and it shows $p=(1+c,1+b+d)$ and $q=(1+a+c,1+d)$. These determine $a,b,c,d$ and differ in both coordinates, so no axis-aligned rectangle shows $\{p,q\}$.
\end{proof}

\begin{lemma}[axis-aligned rivals]\label{lem:rival}
Let no axis-aligned rectangle be fully black. A \emph{rival} of a set $P$ of at most two white points is an axis-aligned rectangle that shows exactly $P$. The empty set has no rival, and $P$ has one exactly when it is a singleton $\{(x,r)\}$ with $\{r,s\}\in F$ for some $s\in B_x$, a vertical pair $\{(x,r),(x,s)\}$ with $\{r,s\}\in F$, a horizontal pair $\{(x,r),(z,r)\}$ with $B_x\cap B_z\ne\emptyset$, or an oblique pair $\{(x,r),(z,s)\}$ with $x\ne z$, $r\ne s$, and $(x,s),(z,r)\in S$.
\end{lemma}

\begin{proof}
A rival of $P\ne\emptyset$ has rows $\{r,s\}$, columns $\{x,z\}$, and its other corners black. As $(x,r)$ is white, this is the listed condition, which forces $z\ne x$ and $s\ne r$.
\end{proof}
\section{Three rows}\label{sec:three}

Throughout, $m\ge3$, the rows are $1,2,3$ from the top, and the tilted rectangles are the
diamonds $D(i,1)$ (Lemma~\ref{lem:tilted-narrow}). Theorem~\ref{thm:rowbound} and
Lemma~\ref{lem:cross} give $f(3,m)=m+2$. This is the result of~\cite{trureturing2026}. The columns are $a=\{1\}$, $b=\{2\}$, $c=\{3\}$,
$A=\{1,2\}$, $B=\{1,3\}$, $C=\{2,3\}$ and $X=\{1,2,3\}$, with $a$ and $c$ \emph{outer}. A
blackout without empty columns is its \emph{word} $B_1\cdots B_m$, and each pair of $F$
(Section~\ref{sec:conv}) is named by its double letter.

\begin{proposition}[the two types]\label{prop:threeclass}
Let $S$ be a blackout of the $3\times m$ grid with $m+2$ points.
\begin{enumerate}[(a)]
\item If $S$ is valid, its word is one $X$ among singletons (type T) or two different doubles
among singletons (type P).
\item For every word of type T or P, $G_S$ is a tree, so axis-aligned rectangles have
distinct presentations.
\item There are $m3^{m-1}+\binom m2\cdot6\cdot3^{m-2}=m^23^{m-1}$ words of type T or P.
\end{enumerate}
\end{proposition}

\begin{proof}
(a) By Lemmas~\ref{lem:H} and~\ref{lem:mantel}, $|S|=t+e(H)$ with $t\le m$ nonempty columns and
$e(H)\le2$. So every column is nonempty and $\sum_i(|B_i|-1)=2$, and two equal doubles would form
a four-cycle in $G_S$, against Lemma~\ref{lem:girth}. (b) The column $X$, or the two doubles,
connect the three rows, and every other column hangs on one row. (c) Place $X$, or two
different doubles, and choose a singleton for each other column.
\end{proof}

These words are the \emph{candidates}, in the class $X$ and the classes $AC$, $CA$, $AB$,
$BA$, $BC$ and $CB$ named by their doubles from left to right. Let $AC_m$ hold the valid words
of width $m$ in $AC$, and likewise for the others.

\begin{lemma}[diamond test]\label{lem:diamond}
A candidate $S$ with pair set $F$ is valid if and only if every diamond $D(i,1)$ passes the
test below, where $P=D(i,1)\setminus S$.
\begin{enumerate}[(a)]
\item If $|P|\ge3$ or $P$ is empty, the diamond passes.
\item If $P=\{(i+1,1),(i+1,3)\}$, the diamond fails if and only if $B\in F$.
\item If $P=\{(i,2),(i+2,2)\}$, it fails if and only if $B_i\cap B_{i+2}$ meets $\{1,3\}$.
\item If $P$ is an oblique pair $\{(x,r),(z,s)\}$, it fails if and only if $(x,s),(z,r)\in S$.
\item If $P=\{(x,r)\}$, it fails if and only if $\{r,s\}\in F$ for some $s\in B_x$.
\end{enumerate}
\end{lemma}

\begin{proof}
Axis-aligned rectangles are told apart (Proposition~\ref{prop:threeclass}(b)), three points fix
a rectangle (Lemma~\ref{lem:threecorners}), and an empty diamond has no axis-aligned rival, as
$G_S$ has no four-cycle. Tests (b) to (e) are Lemma~\ref{lem:rival}, a rival needing $B\in F$
in (b) and $\{r,s\}\in F$ in (e). Two diamonds share at most a tip, so a common presentation
would need both middle columns black in rows $1$ and $3$, which no candidate has.
\end{proof}

A \emph{window} is three consecutive letters, at columns $i$ to $i+2$, and \emph{passes} or
\emph{fails} with $D(i,1)$. Its \emph{tips} $(i,2)$ and $(i+2,2)$ are black
exactly when their letters are $b$, $A$, $C$ or $X$.

\begin{lemma}[singleton dichotomy]\label{lem:dichotomy}
In a candidate, a window of three singletons fails if and only if $B\in F$ and it is
$bab$, $bbb$ or $bcb$.
\end{lemma}

\begin{proof}
With both tips black, the diamond shows the vertical pair or one point of the middle column, and
tests (b) and (e) fail it exactly when $B\in F$. With one black tip it shows three corners or an
oblique pair passing test (d), as $(i+1,2)$ is white. With no black tip at least three corners
show.
\end{proof}

\begin{lemma}[six of nine in the class $\{A,C\}$]\label{lem:sixofnine}
Let $S$ be a candidate with $F=\{A,C\}$, and let $s,t$ be singletons.
\begin{enumerate}[(a)]
\item A window $sAt$ fails if and only if $st$ is $bb$, $bc$ or $cb$, also for $t=C$ read
as $t=b$.
\item A window $sCt$ fails if and only if $st$ is $bb$, $ba$ or $ab$.
\item A window with one double, at a tip, passes.
\end{enumerate}
\end{lemma}

\begin{proof}
(a) Column $i+1$ shows only $(i+1,3)$, and the tip $(i+2,2)$ is black for $t\in\{b,C\}$.
With both tips black, test (e) fails, as $2\in B_{i+1}$ and $C\in F$. With only $(i+2,2)$
black, the diamond shows $(i,2)$ and $(i+1,3)$, and test (d) fails it exactly when $s=c$. The case of $(i,2)$ alone is symmetric, and with no black tip three
corners show. Reflection in the middle row gives (b). (c) For $Ast$ the proof of
Lemma~\ref{lem:dichotomy} applies with the tip $(i,2)$ in place of $s=b$, as $B\notin F$.
Reflection and reversal give the other windows.
\end{proof}

By Lemma~\ref{lem:diamond}, the set $AC_m$ holds the words $UAVCW$ of width $m$, with $U$, $V$
and $W$ singleton strings, whose windows pass. Put
$G_{AC}(z)=\sum_{m\ge2}|AC_m|z^m$, define $G_{AB}$ likewise from the words $UAVBW$, and let
$G_X(z)=\sum_{m\ge1}|X_m|z^m$ come from the words $UXW$. With $R=1-3z$, nonempty singleton
strings with a fixed letter next to a double have generating function $z/R$.

\begin{proposition}[the class $\{A,C\}$]\label{prop:threeGAC}
The generating function of $AC$ is
\begin{equation}\label{eq:GAC}
G_{AC}(z)=\frac{z^2-4z^3+10z^4-8z^5-3z^6}{(1-3z)^3},
\end{equation}
and
\begin{equation}\label{eq:GACcoef}
|AC_m|=2(m-1)(3m+2)3^{m-5}\qquad(m\ge4).
\end{equation}
\end{proposition}

\begin{proof}
By Lemma~\ref{lem:sixofnine}(a) and the half turn, a window $tAC$ passes only for $t=a$ and
$ACt$ only for $t=c$, and $AtC$ passes by tests (b) and (e). With Lemmas~\ref{lem:dichotomy}
and~\ref{lem:sixofnine}(c), only windows with a double in the middle can fail. If $V$ is
empty, a nonempty $U$ ends in $a$ and a nonempty $W$ begins with $c$, giving $z^2(1+z/R)^2$.
If $V=t$, Lemma~\ref{lem:sixofnine}(a),(b) leaves $p_t=3,1,2$ last letters for $U$ and
$q_t=2,1,3$ first letters for $W$, for $t=a,b,c$. As $\sum p_t=\sum q_t=6$ and $\sum
p_tq_t=13$, this gives $z^3\sum_t(1+p_tz/R)(1+q_tz/R)=z^3(3+12z/R+13z^2/R^2)$. If
$|V|\ge2$, each end gives $3+6z/R$, for $v_1$ alone or one of the six passing pairs
$(u,v_1)$, and the middle $1/R$, for $9z^4(1-z)^2/R^3$. The three cases sum
to~\eqref{eq:GAC}, and $[z^m]z^kR^{-3}=\binom{m-k+2}23^{m-k}$ gives~\eqref{eq:GACcoef}.
\end{proof}

\begin{theorem}[the number of three-row optima]\label{thm:threecount}
Let $E_m$ count the optima of the $3\times m$ grid with a column black in rows $1$ and $3$.
For $m\ge4$,
\begin{equation}\label{eq:threecount}
a_3(m)=4(m-1)(3m+2)3^{m-5}+E_m.
\end{equation}
\end{theorem}

\begin{proof}
Only $B$ and $X$ are black in rows $1$ and $3$, so by
Proposition~\ref{prop:threeclass}(a) these optima are those outside $AC$ and $CA$. Column reversal maps
$AC$, $AB$ and $BC$ onto $CA$, $BA$ and $CB$, and the reflection in the middle row maps $AB$
onto $CB$, so
\begin{equation}\label{eq:classsum}
a_3(m)=2|AC_m|+4|AB_m|+|X_m|=2|AC_m|+E_m\qquad(m\ge3),
\end{equation}
and~\eqref{eq:GACcoef} gives~\eqref{eq:threecount}.
\end{proof}

\subsection{The generating function}

Put $L=1-2z-2z^2$, $N=1+2z^2$ and $D=LN$.

\begin{theorem}[three-row generating function, parts (b) and (c) computer-assisted]\label{thm:threegf}
\leavevmode
\begin{enumerate}[(a)]
\item The series $\sum_{m\ge3}a_3(m)z^m$ equals $2G_{AC}+4G_{AB}+G_X-z-12z^2$, with
$G_{AB}$ and $G_X$ from Propositions~\ref{prop:GAB} and~\ref{prop:GX}, of denominators
dividing $D^3$ and $D^2$.
\item Its reduced denominator is exactly $Q=R^3L^3N^3$.
\item With
\begin{align*}
P(z)={}&z-3z^2-34z^3+228z^4-641z^5+1307z^6-2186z^7\\
&+2056z^8-236z^9-1060z^{10}+2584z^{11}-4264z^{12}\\
&-1440z^{13}+704z^{14}-1024z^{15}+2080z^{16}+2176z^{17}+384z^{18},
\end{align*}
we have
\begin{equation}\label{eq:PQ}
\sum_{m\ge3}a_3(m)z^m=\frac{P(z)}{Q(z)}-z-12z^2.
\end{equation}
\end{enumerate}
\end{theorem}

\begin{proof}
(a) Use~\eqref{eq:classsum}, with $z+12z^2$ removing widths one and two. (b) This is
Theorem~\ref{thm:denominator}. (c) Exact computer algebra in SymPy~\cite{sympy2017} gives $P$,
and an independent transfer matrix confirms~\eqref{eq:PQ} exactly through width $100$ (\nolinkurl{three_row_closed_form_check.py}).
\end{proof}

Let $\rho=1+\sqrt3$ and $\sigma=1-\sqrt3$.

\begin{theorem}[three-row closed form, computer-assisted exact algebra]\label{thm:threeclosed}
For every $m\ge3$,
\begin{equation}\label{eq:Eclosed}
E_m=p_+(m)\rho^m+p_-(m)\sigma^m+T_m,
\end{equation}
where $432\,p_\pm(m)=(19\pm11\sqrt3)m^2+(61\pm41\sqrt3)m+104\pm28\sqrt3$ and $T_m=(-2)^{\lfloor m/2\rfloor}\,\kappa(m)$ with $\kappa(m)=(2m^2+41m-104)/216$ for even $m$ and
$\kappa(m)=(7m^2-32m-175)/432$ for odd $m$. Hence $E_m\sim\frac{19+11\sqrt3}{432}m^2\rho^m$. Without the computer, we have $E_m>0$ and
\begin{equation}\label{eq:asymp1}
a_3(m)=\frac4{81}m^23^m-\frac4{243}(m+2)3^m+O(m^2\rho^m).
\end{equation}
\end{theorem}

\begin{proof}
By~\eqref{eq:classsum}, the number $E_m$ is the coefficient of $z^m$ in $4G_{AB}+G_X$, whose
denominator divides $L^3N^3$. Its poles $1/\rho$, $1/\sigma$ and $\pm1/(i\sqrt2)$ are at most
triple, so beyond a polynomial part of degree one, the sequence has the form~\eqref{eq:Eclosed}.
Exact computer algebra gives the twelve rational coefficients. Both sides satisfy the
order-$12$ recurrence of $L^3N^3$ from $m=14$ on and agree for $m=2,\dots,13$, hence always. As $|\sigma|<1$
and $|T_m|=O(m^22^{m/2})$, the asymptotic of $E_m$ follows. Without the computer, the pole
$1/\rho$ of least modulus gives $E_m=O(m^2\rho^m)$, hence~\eqref{eq:asymp1}
by~\eqref{eq:threecount}, and the cross of Lemma~\ref{lem:cross}, with word $Xa\cdots a$,
gives $E_m>0$.
\end{proof}
\subsection{\texorpdfstring{The classes $AB$ and $X$}{The classes AB and X}}\label{app:threegf}

This subsection proves Propositions~\ref{prop:GX} and~\ref{prop:GAB} and
Theorem~\ref{thm:denominator}, which the proof of Theorem~\ref{thm:threegf} uses. Here
$D=LN=1-2z-4z^3-4z^4$. A
\emph{restricted run} is a string of singletons with no two letters $b$ at distance two.
Reversal and reflection carry the next lemma to $BA$, $BC$ and $CB$.

\begin{lemma}[forbidden windows in the classes containing $B$]\label{lem:threewindows}
In the class $AB$ and the triple class, a window fails exactly when the table lists it for its
class, with $s$ any singleton.
\begin{center}\small
\begin{tabular}{@{}lll@{}}
\toprule
class & windows & forbidden \\
\midrule
$AB$, $X$ & singletons only & $bsb$ \\
$AB$ & one double & $bsA$, $Asb$, $bAb$, $bAc$, $cAb$, $aBa$, $aBb$, $bBa$, $cBc$ \\
 & two doubles & $bAB$, $ABa$ \\
$X$ & one triple & $bsX$, $Xsb$, $aXa$, $aXb$, $bXa$, $bXc$, $cXb$, $cXc$ \\
\bottomrule
\end{tabular}
\end{center}
\end{lemma}

\begin{proof}
Singleton windows follow from Lemma~\ref{lem:dichotomy}, and $sAt$ as in
Lemma~\ref{lem:sixofnine}(a), except that $bAb$ now fails test (e) of Lemma~\ref{lem:diamond}
as $B\in F$. In $sBt$ and $sXt$ both corners in column $i+1$ are black, so the
diamond is empty for $s=t=b$, and for outer $s$ and $t$ test (c) fails it exactly when $s=t$.
For $s=b\ne t$ it shows $(i+2,2)$ alone, and test (e) fails it exactly when $\{2,t\}\in F$. In
$uAB$ test (d) fails $u=b$, as $(i+2,3)$ and $(i+1,2)$ are black, and otherwise three corners
show. In $ABw$ the diamond is empty for $w=b$, and test (e) fails it exactly for $w=a$. In $Asb$
and $Xsb$ tests (b) and (e) fail the vertical pair or the point of column $i+1$. Other windows
with a double or $X$ at the left tip show three corners or an oblique pair needing the white
point $(i+1,2)$. The rest is symmetric.
\end{proof}

Give an outer letter bit $0$ and weight $2$, and give $b$ bit $1$. Let $h_{ij}$ count, by
length, the continuations of a restricted run ending in bits $i,j$, where $b$ cannot follow
$1,j$. Read runs $w_1\cdots w_n$ with $n\ge1$
outward from a double. Let $g_0$ and $g_1$ count those with $w_1$ a prescribed outer letter and
with $w_1=b$, and $u$ and $v$ the same runs with $w_2$, if present, outer. A fictitious letter
before $w_1$, outer for $g$ and $b$ for $u$ and $v$, gives $(g_0,g_1,u,v)=z(h_{00},h_{01},
h_{10},h_{11})$. Solving the linear system for $h$ gives
\begin{equation}\label{eq:runs}
Dg_0=z(1+z+3z^2+2z^3),\ \ Dg_1=z(1+z),\ \ Du=z(1+2z^2+2z^3),\ \ Dv=z.
\end{equation}

\begin{proposition}[the triple class]\label{prop:GX}
With $u$ and $v$ as in~\eqref{eq:runs}, $G_X(z)=z\bigl(1+4u+2v+2u^2+v^2\bigr)$, and its
denominator divides $D^2$.
\end{proposition}

\begin{proof}
In $UXW$ the runs are restricted, and $bsX$ and $Xsb$ make the second letters from $X$ outer.
One nonempty run gives $2u+v$ on either side. With two, the window at $X$ allows only the
neighbor pairs $(a,c)$, $(b,b)$ and $(c,a)$, which give $2u^2+v^2$.
\end{proof}

A word of $AB$ is $UAVBW$ with restricted runs $U$, $V$ and $W$. By
Lemma~\ref{lem:threewindows}, a nonempty $U$ read from $A$ is counted by $u$ or $v$, and a
nonempty $W$ read from $B$ by $g_0$ or $g_1$. Let $\ell_t$ and $r_t$ count the allowed $U$ and
$W$, empty included, when $t$ follows $A$ or precedes $B$. The windows at the
double give
\begin{equation}\label{eq:lr}
(\ell_a,\ell_b,\ell_c)=(1+2u+v,1+u,1+2u),\ (r_a,r_b,r_c)=(1+g_0,1+g_0+g_1,1+g_0+g_1).
\end{equation}
For $s\in\{00,10\}$ let $J_{s\beta}$ count the continuations from $s$ ending in bit $\beta$, the
empty one with $\beta=0$. Solving as for $h$ gives, with rows $s$ and columns $\beta$,
\begin{equation}\label{eq:J}
J=\frac1D\begin{pmatrix}1+2z^2+2z^3&z+z^2\\1&2z^2+2z^3\end{pmatrix}.
\end{equation}

\begin{proposition}[the class $\{A,B\}$, polynomial form computer-assisted]\label{prop:GAB}
With~\eqref{eq:runs} to~\eqref{eq:J},
\begin{equation}\label{eq:GAB}
\begin{aligned}
G_{AB}(z)={}&z^2(1+2u)(1+g_0+g_1)+z^3(\ell_ar_a+\ell_br_b+\ell_cr_c)\\
&+z^4\,(\ell_a+\ell_c,\ \ell_b)\,J\,(r_a+r_c,\ 2r_b)^{T}.
\end{aligned}
\end{equation}
Moreover $G_{AB}=P_B/D^3$ with
\begin{equation}\label{eq:PB}
P_B(z)=z^2(1+z)(1+z^2-3z^3+4z^6+4z^7).
\end{equation}
\end{proposition}

\begin{proof}
Split by the length $k$ of $V$. For $k=0$ the windows $uAB$ and $ABw$ give the first term, as
$u\ne b$ and $w\ne a$. For $k=1$ the window $AtB$ passes, and $uAt$ and $tBw$ give $\ell_tr_t$.
For $k\ge2$ the window $Av_1v_2$ makes $v_2$ outer, $v_{k-1}v_kB$ passes, and $v_3\cdots v_k$
is a continuation from bits $00$ or $10$, as $v_1$ is outer or $b$, with weight
$\ell_{v_1}r_{v_k}$. Since $r_a+r_c$ counts both outer $v_k$, for $k\ge3$ the weight $2$ of an outer $v_k$
in $J$, like the $2$ in $2r_b$, counts the choices of $v_2$, and $k=2$ is the constant term of
$J$.
Exact computer algebra gives~\eqref{eq:PB}.
\end{proof}

\begin{theorem}[the exact denominator, computer-assisted]\label{thm:denominator}
The reduced denominator of $\sum_{m\ge3}a_3(m)z^m$ is $Q=R^3L^3N^3$.
\end{theorem}

\begin{proof}
By Theorem~\ref{thm:threegf}(a) the denominator divides $R^3D^3=Q$. No two of $R$, $L$ and $N$
share a root, as $D(\frac13)\ne0$ and $L-N=-2z-4z^2$ vanishes only where $N\ne0$. At
$z=\frac13$ only $G_{AC}$ has a pole, and the numerator in~\eqref{eq:GAC} is $\frac4{81}$. As $P_B\bmod L=\frac z2$ and $P_B\bmod N=\frac14-\frac z2$, at each root
of $D$ the term $G_{AB}$ has a triple pole, $G_X$ at most a double pole, and $G_{AC}$ none.
\end{proof}

\begin{remark}[optima up to symmetry]\label{rem:threeorbits}
Let $o_3(m)$ count optima up to the reflection in the middle row and the column reversal. It
is $15$, $58$, $227$, $864$, $3190$ and $11502$ at widths $3$ to $8$. By Burnside's lemma,
$4o_3(m)$ is $a_3(m)$ plus the optima fixed by the reflection, the reversal and the half turn.
An optimum fixed by the reflection uses only $b$, $B$ and $X$, so it is one $X$ among letters
$b$, and for $m\ge4$ a window $Xbb$, $bbX$ or $bbb$ fails (Lemmas~\ref{lem:diamond}(b)
and~\ref{lem:dichotomy}), so only $bXb$ is fixed.

Write $m=2k$ or $m=2k+1$. The half turn reverses the word and swaps $A$ with $C$ and $a$ with
$c$. A fixed optimum with two doubles thus has $A$ and $C$ at mirror positions $j\le k$ and $m+1-j$,
and center $b$ for odd $m$. Its first $k$ letters determine it, and only its windows centered
at $A$ or $C$ can fail (proof of Proposition~\ref{prop:threeGAC}), those at $C$ mirroring
those at $A$. By Lemma~\ref{lem:sixofnine}(a), this leaves $3^{k-1}$ words in $AC$ for $j=1$,
$3^{k-2}$ for $j=k\ge2$, $2\cdot3^{k-2}$ for $1<j<k$, and as many in $CA$. So for $k\ge2$
there are $4k3^{k-2}$ of them, and for $m=3$ there are $AbC$ and $CbA$, with series
$W=2z^3+\frac49(1+z)\bigl(3z^2(1-3z^2)^{-2}-3z^2\bigr)$.

A palindrome repeats each letter off its center, so the reversal fixes only words $UXU^{\mathrm{rev}}$ of odd width. There $tXt$ passes only for $t=b$, and
$bsX$ and $Xsb$ make the letter at distance two from $X$ outer
(Lemma~\ref{lem:threewindows}), so $U$ read outward is counted by $v$ in~\eqref{eq:runs}. In
the words $UX\iota(U^{\mathrm{rev}})$ fixed by the half turn, with $\iota$ the reflection, the
central pairs $(a,c)$, $(b,b)$ and $(c,a)$ pass, so $U$ is counted by $2u+v$. Adding $z^3$ for
$bXb$, $2z\bigl(u(z^2)+v(z^2)\bigr)$ and $W$ gives
\begin{align*}
4\sum_{m\ge3}o_3(m)z^m={}&\sum_{m\ge3}a_3(m)z^m+z^3+\frac{4z^3(1+z^4+z^6)}{M(1+2z^4)}\\
&+\frac{2z^3(1-z)(1+5z+3z^2-3z^3)}{(1-3z^2)^2},
\end{align*}
with $M=1-2z^2-2z^4$, since $D(z^2)=M(1+2z^4)$. A computer check
(\nolinkurl{three_row_orbits.py}) confirms the fixed counts through width $24$ and $o_3(m)$ for
$m\le10$.
\end{remark}
\section{Four rows}\label{sec:four}

\begin{theorem}[four rows]\label{thm:fourvalue}
For every $m\ge6$ we have $f(4,m)=m+4$.
\end{theorem}

Theorem~\ref{thm:rowbound} gives the upper bound. For the lower bound take
\begin{equation}\label{eq:fourS}
S=\{(i,1):2\le i\le m-1\}\cup\{(m-1,2),(m,2),(1,3),(2,3),(1,4),(m,4)\}.
\end{equation}
Here $|S|=m+4$ and $G_S$ is the eight-cycle $r_1\,c_2\,r_3\,c_1\,r_4\,c_m\,r_2\,c_{m-1}$ with leaves $c_3,\ldots,c_{m-2}$ on $r_1$.

\begin{proof}[Proof of Theorem~\ref{thm:fourvalue}]
As $G_S$ has no four- or six-cycle, axis-aligned rectangles have distinct presentations and none is hidden, by Lemma~\ref{lem:girth}.

By Lemma~\ref{lem:tilted-narrow}, tilted corners are black in row $1$ and white in row $4$. The other black corners are $(i+2,2)$ if $i\ge m-3$ and $(i+1,3)$ if $i=1$ in $D(i,1)$, $(i,3)$ if $i\le2$ and $(i+1,2)$ if $i=m-2\ge4$ in $D(i,2)$, and $(i,3)$ if $i\le2$ and $(i+3,2)$ if $i\ge m-4$ in $\kA_i$ and $\kB_i$. By Lemma~\ref{lem:threecorners} a rectangle with at most one black corner is safe, leaving those below, where $s=1$ for $\kA_i$ and $s=2$ for $\kB_i$.
\begin{center}\small
\begin{tabular}{@{}lll@{}}
\toprule
rectangle & which $i$ & corners shown\\
\midrule
$D(1,1)$ & & $(1,2),(3,2)$\\
$D(i,1)$ & $i\in\{m-3,m-2\}$ & $(i,2),(i+1,3)$\\
$\kA_i$ or $\kB_i$ & $i\in\{1,2\}$, $i\notin\{m-4,m-3\}$ & $(i+s,4),(i+3,2)$\\
$\kA_i$ or $\kB_i$ & $i\in\{m-4,m-3\}$, $i\notin\{1,2\}$ & $(i,3),(i+s,4)$\\
$\kA_i$ or $\kB_i$ & $i=2$, $m=6$ & $(i+s,4)$\\
\bottomrule
\end{tabular}
\end{center}

Shown rows differ between the first four lines, and within a line the point in row $2$ or $3$ fixes $i$ and that in row $4$ fixes $s$.

No row $r$ has $(1,r)$ and $(3,r)$ black, so $\{(1,2),(3,2)\}$ has no rival. In the next three lines the axis-aligned rectangle through the shown points has white corner $(i,3)$, $(i+s,2)$, or $(i,4)$, respectively, since $i\ge3$, $i+s\le4<m-1$, and $3\le i\le m-3$ there. In the last line $m=6$, and the shown point $(z,4)$ has $z\in\{3,4\}$ and no rival, as column $z$ is black only in row $1$ and no column is black in rows $1$ and $4$.
\end{proof}

By Theorem~\ref{thm:equality}, for $m\ge4$ the axis-valid blackouts with $m+4$ points are the $3\,(m)_4\,4^{m-4}$ \emph{candidates}, with four doubles whose row pairs form a four-cycle $F$ and singletons elsewhere.

\begin{corollary}[widths four and five, computer-assisted]\label{cor:fourboundary}
We have $f(4,4)=7$ and $f(4,5)=8$.
\end{corollary}

\begin{proof}
Lemma~\ref{lem:cross} gives the lower bounds, and a computer finds a repeated presentation in all $72$ and $1{,}440$ candidates.
\end{proof}

\begin{theorem}[no hidden rectangle]\label{thm:hidden}
For $m\ge4$, no valid blackout of the $4\times m$ grid hides a rectangle.
\end{theorem}

\begin{proof}
By Lemma~\ref{lem:diff} it suffices to write each corner set as $C(R_1)\triangle C(R_2)$ with $R_1\ne R_2$. For an axis-aligned rectangle, keep its rows and pair each of its columns $x,y$ with a third column $w$. For $1\le i\le m-3$ let $\mathsf Q_i=\{(i,2),(i,3),(i+3,2),(i+3,3)\}$. Then
\begin{equation}\label{eq:fourxor}
\begin{gathered}
\kC_i=\kA_i\,\triangle\,\mathsf Q_i,\qquad \kE_i=\kB_i\,\triangle\,\mathsf Q_i,\\
D(i+1,1)=D(i,2)\,\triangle\,\kA_i,\qquad D(i+1,2)=D(i,1)\,\triangle\,\kE_i,
\end{gathered}
\end{equation}
and each solves for any of its three rectangles. The first line covers $\kA_i$, $\kB_i$, $\kC_i$, and $\kE_i$, and the second, with $i=j-1$ or $i=j$, covers $D(j,1)$ and $D(j,2)$, since $j\ge2$ or $j\le m-3$ as $m\ge4$.
\end{proof}

\begin{theorem}[forbidden-block bound]\label{thm:foursurvival}
For every $m\ge6$ we have $a_4(m)\le3\,(m)_4\,4^{m-4}(53/64)^k$, where
$k=\max(0,\lfloor m/4\rfloor-4)$. So $a_4(m)=O\bigl(m^4\,212^{m/4}\bigr)$.
\end{theorem}

\begin{proof}
Optima are candidates, with $3\,(m)_4$ choices of $F$, the double columns, and their
pairs. A \emph{block} is a run of four singleton columns, whose word lists
their black rows. For a block from column $x$, the \emph{template words} $2413$,
$3412$, $2143$, and $3142$ are the corner rows of $\kC_x$, $\kA_x$, $\kE_x$, and $\kB_x$. Let
$T$ be one, with word $t_1t_2t_3t_4$.

A \emph{swap word} swaps two letters $t_p$ and $t_q$. No block carries one, since $T$ and
the axis-aligned rectangle on $(x+p-1,t_p)$ and $(x+q-1,t_q)$ would both show exactly these
two points. Template words are the permutations with both end letters in $\{2,3\}$.
Swapping two end or two middle letters gives one again, and swapping an end
and a middle letter gives the sixteen with one end letter in $\{2,3\}$, so the $20$ swap words are these and the four template words.

An \emph{edit word} replaces a letter $t_p=r$ by $s$ with $\{r,s\}\in F$. No block carries
one, since $T$ and the rectangle on rows $r,s$ and columns $x+p-1$ and $d$, the double column
of $\{r,s\}$, would both show only $(x+p-1,r)$. Each template word has $8$ edit words.
Two template words share one only if they differ in two positions holding $r,s$ in opposite
orders, and then share two if $\{r,s\}\in F$, which happens for two pairs with
$\{r,s\}=\{2,3\}$ and two with $\{1,4\}$. So there are at least $32-8=24$ edit words. They repeat a letter, so none is a swap word, and at most
$256-20-24=212=4^4\cdot53/64$ words are allowed in a block.

At most four of the $\lfloor m/4\rfloor$ column groups $4j-3,\ldots,4j$ contain a double
column, so at least $k$ are blocks. Other columns have at most $4$ values.
\end{proof}

\begin{lemma}[six columns of memory]\label{lem:fourmemory}
For $m\ge6$ and a pair set $F$, a word over the four singleton letters and the pairs of $F$, each pair used once, is the word of an optimum exactly when (i) no tilted rectangle is hidden, (ii) none shares its presentation with an axis-aligned rectangle whose columns are among its own and the double columns, and (iii) no two tilted rectangles within seven consecutive columns share a presentation.
\end{lemma}

\begin{proof}
Optima satisfy the conditions, (i) by Theorem~\ref{thm:hidden}. Conversely, the word is axis-valid by Theorem~\ref{thm:equality}, and a hidden axis-aligned rectangle would be a four-cycle of $G_S$. Given (i), two distinct rectangles with one presentation $P$ have $P\ne\emptyset$, and one is tilted. If the other is axis-aligned, each of its columns meets $P$, so it is a column of the tilted one or, with two black corners, a double column, against (ii). If both are tilted, they share a column $c$ meeting $P$. Each spans at most four consecutive columns, so both lie in columns $c-3$ to $c+3$, against (iii).
\end{proof}

\begin{theorem}[the four-row generating function, computer-assisted]\label{thm:fourgfexact}
There are coprime integer polynomials $P_4=26z^6-826z^7+\cdots$ of degree $444$ and $Q_4=\sum_{j=0}^{420}q_jz^j$ of degree $420$ with $q_0=1$ and $\sum_{m\ge6}a_4(m)z^m=P_4/Q_4$. The recurrence $\sum_jq_j\,a_4(n-j)=0$ holds for every $n\ge445$ and fails at $n=444$. Over the integers
\[
Q_4=(1-z)^2(1+z)^2\,g_5^3\,\tilde g_5^3\,g_6^3\,g_{11}^5\,g_{17}^3\,g_{19}^3\,g_{20}^5\,g_{21}^5,
\]
where $g_d$ and $\tilde g_d$ are irreducible of degree $d$ with constant term $1$.
\end{theorem}

\begin{proof}
By Lemmas~\ref{lem:fourmemory} and~\ref{lem:rival}, an automaton remembering the last six columns and the used pairs of $F$ accepts exactly the optima with pair set $F$. Two transfer counters built from the definition~\cite{repo}, one checked against this automaton, agree on $a_4(m)$ for $m\le1500$ and with Table~\ref{tab:values}. Berlekamp--Massey modulo eight primes gives $P_4$ and $Q_4$. With a register of used pairs, the automaton has a transfer matrix $T_F$ for each of the three pair sets $F$, the four-cycles with parts $\{1,2\}$ and $\{3,4\}$, $\{1,3\}$ and $\{2,4\}$, or $\{1,4\}$ and $\{2,3\}$, with $2{,}519$, $3{,}089$, and $2{,}119$ reachable states. Its state vectors $v(n)=v(0)T_F^n$ satisfy $\sum_jq_jv(n_F-j)=0$ at $n_F=436$, $435$, and $445$, so powers of $T_F$ give the recurrence for $n\ge445$, and $\deg P_4=444$ forces its failure at $n=444$. SymPy factors $Q_4$.
\end{proof}

\begin{corollary}[growth, computer-assisted]\label{cor:fourgrowth}
Let $\rho_4=3.351996777310\ldots$ be the largest real root of $p(x)=x^{20}g_{20}(1/x)=x^{20}-4x^{19}+2x^{18}+\cdots-26x^2-4x+8$. There is a polynomial $B$ of degree $4$ with $a_4(m)=\rho_4^m B(m)+O((1000/341)^m)$, so $a_4(m)\sim c\,m^4\rho_4^m$ with $c\approx8.1611\times10^{-5}$.
\end{corollary}

\begin{proof}
Exact root isolation on the factors of $Q_4$ shows that $r=1/\rho_4=0.29832\ldots$ is its only root of modulus at most $0.341$, simple in $g_{20}$ and so of order five. So $P_4/Q_4=\sum_{j=1}^{5}c_j(1-z/r)^{-j}+E(z)$, where $E$ has coefficients $O((1000/341)^m)$ and, by coprimality, $c_5=H(r)\ne0$ for $H(z)=P_4(z)(1-z/r)^5/Q_4(z)$. As $[z^m](1-z/r)^{-j}=\rho_4^m\binom{m+j-1}{j-1}$, we get $B(m)=\sum_jc_j\binom{m+j-1}{j-1}$ and $c=c_5/24$. Interval arithmetic puts $c$ between $8.1610970775\times10^{-5}$ and $8.1610970776\times10^{-5}$.
\end{proof}
\section{Five and six rows}\label{sec:five}

\subsection{Boundary padding}

For a blackout $S$ of the $h\times m$ grid and $L,R\ge0$, let $S_{L,R}$ add the columns
$1-L,\dots,0$ and $m+1,\dots,m+R$, each black exactly in the \emph{padding row}, row $1$ or
row $h$.

\begin{theorem}[finite collar]\label{thm:collar}
Let $h\ge3$, fix the padding row and the padded sides, and let $L^*$ and $R^*$ be $W_h$ of
Lemma~\ref{lem:span} on padded sides and $0$ otherwise. Then $S_{L,R}$ is valid for all
$L,R\ge0$ vanishing on unpadded sides if and only if $S_{L^*,R^*}$ is valid.
\end{theorem}

\begin{proof}
Write $S^*=S_{L^*,R^*}$. One direction is clear. Let $S^*$ be valid and $U=S_{L,R}$. By
Lemma~\ref{lem:leftmost} and \eqref{eq:tilt-param}, a tilted rectangle has at most one corner
in row $1$ and at most one in row $h$. Hence a weak tilted rectangle of $U$ has a black corner in an original column $e$,
and by Lemma~\ref{lem:span} its columns lie in
$[e-W_h,e+W_h]\cap[1-L,m+R]\subseteq[1-L^*,m+R^*]$. As $U$ and $S^*$ agree on their common
columns, it is a rectangle of $S^*$ with the same presentation. Suppose distinct
rectangles $R_1$ and $R_2$ of $U$ have the same presentation $P$.

\emph{Step 1. Both axis-aligned.} Added columns are vertices of degree one, so the cycles of
$G_U$ lie in $G_{S^*}$. As $S^*$ is valid with at least three columns, $U$ is axis-valid by
Lemma~\ref{lem:girth}.

\emph{Step 2. Both tilted.} By Lemma~\ref{lem:threecorners}, $|P|\le2$, so both are weak, hence
rectangles of $S^*$ with the same presentation, a contradiction.

\emph{Step 3. One of each.} Let $R_1$ be tilted, hence a rectangle of $S^*$. A column of
$R_2$ that meets $P$ is a column of $R_1$, and one that does not has two black corners and is
an original column. So $R_2$ is a rectangle of $S^*$ too, a contradiction.
\end{proof}

\section{Every fixed height}\label{sec:height}

Fix $h$. A blackout of the $h\times m$ grid is a \emph{column word} $w=B_1B_2\cdots B_m$ over the subsets of $\{1,\dots,h\}$, with length $|w|$ and blackout $S(w)$. Let $\mathcal L_h$ be the set of words with a valid blackout, the empty word included.

\begin{theorem}[regularity and rational series]\label{thm:regular}
For every $h\ge1$ the language $\mathcal L_h$ is regular, and $G_h(z,u)=\sum_{w\in\mathcal L_h}z^{|w|}u^{|S(w)|}$ is rational in $z$ and $u$.
\end{theorem}

\begin{proof}
The \emph{trace} of a rectangle $R$ is the word whose $i$th letter is the set of rows of its corners in column $i$, and it determines $C(R)$. By Lemma~\ref{lem:diff}, $w\notin\mathcal L_h$ exactly when two different traces $Y,Y'$ of length $|w|$ have $Y_i\triangle Y'_i\subseteq B_i$ for every $i$. The traces form a regular language, $\emptyset^*\{r,s\}\emptyset^*\{r,s\}\emptyset^*$ for axis-aligned rectangles and, by Lemmas~\ref{lem:leftmost} and~\ref{lem:span}, $\emptyset^*Y_0Y_1\cdots Y_n\emptyset^*$ for finitely many templates \eqref{eq:tilt-param} with $n\le W_h$. So an automaton that reads $w$ and two traces in parallel recognizes the complement of $\mathcal L_h$. Weighting a letter $B$ by $u^{|B|}$, the transfer matrix $M(u)$ of a deterministic automaton for $\mathcal L_h$ gives $G_h=\alpha(I-zM(u))^{-1}\beta$, rational by Cramer's rule \cite[Proposition~V.6]{flajolet2009}.
\end{proof}

\begin{theorem}[fixed height]\label{thm:periodic}
Let $h\ge2$. (a) There are integers $m_0\ge3$, $p\ge1$, and $k_0,\dots,k_{p-1}$ such that $f(h,m)=m+k_{m\bmod p}$ for every $m\ge m_0$. (b) The series $\sum_{m\ge0}a_h(m)z^m$ is rational, with the conventions of Section~\ref{sec:conv}, and the words of optimum blackouts form a regular language.
\end{theorem}

\begin{proof}
Let $E(w)=|S(w)|-|w|$. By Theorem~\ref{thm:rowbound} and $|S(v)|\le h|v|$, every factor $v=B_i\cdots B_j$ of a valid word has $E(v)\le C_h=\max(q_h,2h-2)$. So a valid $w=uv$ of excess $k$ has $k-C_h\le E(u)\le C_h$. An automaton for $\mathcal L_h$ with a counter for the excess of prefixes, kept in this range, accepts the set $\mathcal L_{h,k}$ of valid words of excess $k$, so the lengths of these words are eventually periodic. For $m\ge3$, Lemma~\ref{lem:cross} and Theorem~\ref{thm:rowbound} make $f(h,m)-m$ the largest $k$ from $h-1$ to $q_h$ with a word of length $m$ in $\mathcal L_{h,k}$, which gives (a). The optimum words are those of $\mathcal L_{h,k_r}$ of lengths $m\ge m_0$ with $m\equiv r\pmod p$, and finitely many shorter ones, so (b) follows as in Theorem~\ref{thm:regular}.
\end{proof}

\section{Square grids}\label{app:square}

On narrow grids $f(h,m)-m$ stays bounded. Here we show that on square grids $f(n,n)$ grows
faster than any multiple of $n$, so the guess $f(h,m)=h+m$ fails badly there.

Let $n\ge2$, $\Lambda=1+\ln n$, $\mathcal R$ the set of rectangles of the $n\times n$ grid, tilted ones included, and $d_v$ the number with corner $v$.

\begin{lemma}\label{lem:compress}
If rectangles $R_1\ne R_2$ have exactly two common corners $p$ and $q$, and $\{p,q\}$ is a side of both or a diagonal of both, then $C(R_1)\triangle C(R_2)$ is the corner set of a rectangle.
\end{lemma}

\begin{proof}
For a common side, the other corners are $p+u,q+u$ and $p+v,q+v$ with $u\ne v$ nonzero and perpendicular to $q-p$, so they are distinct and form a rectangle with sides $q-p$ and $v-u$. For a common diagonal, the other corners are $o\pm u$ and $o\pm v$ with $o=(p+q)/2$, $|u|=|v|$, and $v\ne\pm u$. The segments $[o-u,o+u]$ and $[o-v,o+v]$ share midpoint and length, so they are diagonals of a rectangle.
\end{proof}

Let $r_2(t)$ count the integer pairs $(a,b)$ with $a^2+b^2=t$, and $\tau(t)$ the divisors of $t$. By Hardy and Wright \cite[Sections~16.9, 16.10, and~18.1]{hardywright2008}, $r_2(t)\le4\tau(t)$, and each $\varepsilon>0$ has a $K_\varepsilon$ with $\tau(t)\le K_\varepsilon t^\varepsilon$ for $t\ge1$.

Let $\mathcal A$ consist of the sets $C(R)$ with $R\in\mathcal R$ and the sets $C(R_1)\triangle C(R_2)$ whose exactly two common corners form a side of one and a diagonal of the other, and $\mathcal B$ of the sets $C(R_1)\triangle C(R_2)$ with exactly one common corner.

\begin{lemma}\label{lem:counts}
\leavevmode
\begin{enumerate}[(a)]
\item Each point $v$ has $d_v\le8n^2\Lambda$, and $|\mathcal R|\le2n^4\Lambda$.
\item For every $\varepsilon>0$, $|\mathcal A|\le A_\varepsilon n^{4+2\varepsilon}\Lambda$ with $A_\varepsilon=2(1+16\cdot2^\varepsilon K_\varepsilon)$, and $|\mathcal B|\le32n^6\Lambda^2$.
\item A blackout that contains no member of $\mathcal A\cup\mathcal B$ is valid.
\end{enumerate}
\end{lemma}

\begin{proof}
(a) The sides at $v$ are $k(a,b)$ and $l(-b,a)$ with $\gcd(a,b)=1$, $k\ge1$, and $l\ne0$. With $d=\max(|a|,|b|)$ there are $8d$ choices of $(a,b)$, at most $(n-1)/d$ of $k$, and $2(n-1)/d$ of $l$. Each rectangle counts twice, so $2d_v\le\sum_{d=1}^{n-1}16(n-1)^2/d\le16n^2\Lambda$, and $4|\mathcal R|=\sum_vd_v\le8n^4\Lambda$.

(b) A rectangle with diagonal $\{p,q\}$ is fixed by its other diagonal, an integer vector of squared length $t=|p-q|^2<2n^2$, so each side of each $R\in\mathcal R$ is a diagonal of at most $r_2(t)\le4K_\varepsilon(2n^2)^\varepsilon$ rectangles. Hence $|\mathcal A|\le|\mathcal R|(1+16K_\varepsilon2^\varepsilon n^{2\varepsilon})\le A_\varepsilon n^{4+2\varepsilon}\Lambda$. Members of $\mathcal B$ come from two rectangles at a corner $v$, so $|\mathcal B|\le\sum_v\binom{d_v}2\le\frac{n^2}2(8n^2\Lambda)^2$.

(c) By Lemma~\ref{lem:diff}, an invalid blackout $S$ contains $C(R_1)\triangle C(R_2)$ for some $R_1\ne R_2$, which share at most two corners by Lemma~\ref{lem:threecorners}. With two, this set lies in $\mathcal A$ by Lemma~\ref{lem:compress} or by definition; with one, in $\mathcal B$; with none, $C(R_1)\subseteq S$ lies in $\mathcal A$.
\end{proof}

\begin{theorem}[deletion method]\label{thm:alteration}
We have
\[
\liminf_{n\to\infty}\frac{f(n,n)\,(1+\ln n)^{2/5}}{n^{6/5}}\ge\frac5{6\cdot192^{1/5}}=0.2911\ldots.
\]
In particular $f(n,n)-2n\to\infty$.
\end{theorem}

\begin{proof}
Following Alon and Spencer~\cite[Section~3.2]{alonspencer2016}, put $a=192^{-1/5}$, $T=n^{6/5}\Lambda^{-2/5}$, and $\theta=a\,n^{-4/5}\Lambda^{-2/5}\in(0,1)$, and select grid points independently with probability $\theta$. Let $X$, $Y$, and $Z$ count the selected points and the fully selected members of $\mathcal A$ and $\mathcal B$. Then
\[
\mathbb E(X-Y-Z)=n^2\theta-|\mathcal A|\theta^4-|\mathcal B|\theta^6 .
\]
Here $n^2\theta=aT$, and Lemma~\ref{lem:counts}(b) with $\varepsilon=1/10$ gives $|\mathcal B|\theta^6\le32a^6T$ and $|\mathcal A|\theta^4\le A_{1/10}a^4n^{-1/5}T$. Since $32a^5=\frac16$, some selection $S'$ has $X-Y-Z\ge(\frac56a-A_{1/10}a^4n^{-1/5})T$. Deleting a point of each member of $\mathcal A\cup\mathcal B$ inside $S'$ leaves at least $X-Y-Z$ points, valid by Lemma~\ref{lem:counts}(c). So $f(n,n)/T\ge\frac56a-A_{1/10}a^4n^{-1/5}\to5/(6\cdot192^{1/5})$, and $T/n\to\infty$ gives $f(n,n)-2n\to\infty$.
\end{proof}
